    62→/** Extract a flat [lng, lat] list from a Polygon or MultiPolygon geometry. */
    63→function extractCoords(geometry: { type: string; coordinates: unknown }): [number, number][] {
    64→  const pts: [number, number][] = [];
    65→  if (geometry.type === 'Polygon') {
    66→    for (const ring of (geometry.coordinates as number[][][])) {
    67→      for (const [lng, lat] of ring) {
    68→        if (Number.isFinite(lng) && Number.isFinite(lat)) pts.push([lng, lat]);
    69→      }
    70→    }
    71→  } else if (geometry.type === 'MultiPolygon') {
    72→    for (const poly of (geometry.coordinates as number[][][][])) {
    73→      for (const ring of poly) {
    74→        for (const [lng, lat] of ring) {
    75→          if (Number.isFinite(lng) && Number.isFinite(lat)) pts.push([lng, lat]);
    76→        }
    77→      }
    78→    }
    79→  }
    80→  return pts;
    81→}

    83→/**
    84→ * Compute bounding box covering the regional boundary, AOI, thermal field, candidates,
    85→ * and the selected location.
    86→ */
    87→function computeBounds(
    88→  analysisAoi: PolygonAOI | null,
    89→  spatialField: PolygonAOI | null,
    90→  candidateLocs: LocationPoint[],
    91→  targetLoc: LocationPoint,
    92→  regionBoundary?: PolygonAOI | null,
    93→  fitToRegionOnly = false,
    94→): [[number, number], [number, number]] | null {
    95→  const allPts: [number, number][] = [];
    96→
    97→  if (fitToRegionOnly && regionBoundary && regionBoundary.features.length > 0) {
    98→    for (const f of regionBoundary.features) {
    99→      allPts.push(...extractCoords(f.geometry as { type: string; coordinates: unknown }));
   100→    }
   101→  } else {
   102→    // Include target location point
   103→    if (Number.isFinite(targetLoc.longitude) && Number.isFinite(targetLoc.latitude)) {
   104→      allPts.push([targetLoc.longitude, targetLoc.latitude]);
   105→    }
   106→
   107→    // Canonical Analysis AOI vertices
   108→    if (analysisAoi && analysisAoi.features.length > 0) {
   109→      for (const f of analysisAoi.features) {
   110→        const pts = extractCoords(f.geometry as { type: string; coordinates: unknown });
   111→        for (const [lng, lat] of pts) {
   112→          if (Math.abs(lng - targetLoc.longitude) < 0.35 && Math.abs(lat - targetLoc.latitude) < 0.35) {
   113→            allPts.push([lng, lat]);
   114→          }
   115→        }
   116→      }
   117→    }
   118→
   119→    // FortyGuard thermal field vertices
   120→    if (spatialField && spatialField.features.length > 0) {
   121→      for (const f of spatialField.features) {
   122→        const pts = extractCoords(f.geometry as { type: string; coordinates: unknown });
   123→        for (const [lng, lat] of pts) {
   124→          if (Math.abs(lng - targetLoc.longitude) < 0.35 && Math.abs(lat - targetLoc.latitude) < 0.35) {
   125→            allPts.push([lng, lat]);
   126→          }
   127→        }
   128→      }
   129→    }
   130→
   131→    // Candidate coordinates near targetLoc
   132→    for (const { longitude, latitude } of candidateLocs) {
   133→      if (Number.isFinite(longitude) && Number.isFinite(latitude)) {
   134→        if (Math.abs(longitude - targetLoc.longitude) < 0.35 && Math.abs(latitude - targetLoc.latitude) < 0.35) {
   135→          allPts.push([longitude, latitude]);
   136→        }
   137→      }
   138→    }
   139→  }
   140→
   141→  if (allPts.length === 0) return null;
   142→
   143→  let minLng = allPts[0][0], maxLng = allPts[0][0];
   144→  let minLat = allPts[0][1], maxLat = allPts[0][1];
   145→  for (const [lng, lat] of allPts) {
   146→    if (lng < minLng) minLng = lng;
   147→    if (lng > maxLng) maxLng = lng;
   148→    if (lat < minLat) minLat = lat;
   149→    if (lat > maxLat) maxLat = lat;
   150→  }
   151→
   152→  const padRatio = fitToRegionOnly ? 0.08 : 0.25;
   153→  const padLng = Math.max(0.001, (maxLng - minLng) * padRatio);
   154→  const padLat = Math.max(0.001, (maxLat - minLat) * padRatio);
   155→
   156→  return [
   157→    [minLng - padLng, minLat - padLat],
   158→    [maxLng + padLng, maxLat + padLat],
   159→  ];
   160→}

   162→/** Return true if the spatial field contains at least one feature with a valid temperature. */
   163→function hasRenderableTemperatureData(aoi: PolygonAOI | null | undefined): boolean {
   164→  return !!aoi && aoi.features.length > 0 && aoi.features.some((f) =>
   165→    Number.isFinite(Number(f.properties?.average_temperature))
   166→  );
   167→}
