    88→async function safeJsonFetch<T = Record<string, unknown>>(
    89→  url: string,
    90→  init?: RequestInit
    91→): Promise<{ ok: boolean; data: T | null }> {
    92→  try {
    93→    const res = await fetch(url, init);
    94→    const text = await res.text();
    95→    if (!text || !text.trim()) return { ok: res.ok, data: null };
    96→    try {
    97→      return { ok: res.ok, data: JSON.parse(text) as T };
    98→    } catch {
    99→      return { ok: false, data: null };
   100→    }
   101→  } catch {
   102→    return { ok: false, data: null };
   103→  }
   104→}

   423→        category: 'VALIDATION',
   424→      });

   426→    }

   452→        category: 'COVERAGE',
   453→      });
   454→      return;
   455→    }
   456→
   457→    // DEMO canonical AOI: the CAPTURED REQUEST AREA (the genuine polygon_aoi
   458→    // the capture was requested with — mirrored from the fixture metadata).
   459→    // The captured field is replayed verbatim inside it; DEMO never submits a

   471→        // user-selected provider granularity.
   472→        granularity: dataSourceMode === 'FIXTURE' ? FIXTURE_DISPLAY_GRANULARITY : prefs.analysisResolution,
   473→        analysisAreaShape: prefs.analysisAreaShape,
   474→        // Canonical analysis AOI — the EXACT geometry rendered on the map
   475→        // (including any drag the user performed).
   476→        analysisAoi: requestAoi,
   477→        temporalInput: temporal,
   478→        timezone: tz,
   479→      };
   480→
   481→      // FIXTURE mode sends NO candidates — the server uses the actual three
   482→      // captured Manhattan sites. LIVE sends the user's real candidates.
   483→      if (dataSourceMode === 'LIVE' && candidates && candidates.length > 0) {
   484→        body.candidates = candidates.map((c) => ({
   485→          locationId: c.locationId,
   486→          name: c.name,
   487→          latitude: c.location.latitude,
   488→          longitude: c.location.longitude,
   489→        }));
   490→      }
   491→
   492→      const { ok, data } = await safeJsonFetch<{
   493→        success: boolean;
   494→        decision?: DecisionResult;
   495→        spatialDecision?: SpatialDecisionResult;
   496→        jointDecision?: JointDecisionResult;

  1017→  //   EMPTY → LOCATION_SELECTED → AOI_VALID → READY → GENERATING → RESULT,
  1018→  //   with AOI_INVALID (retained visibly + Generate disabled) and ERROR /
  1019→  //   NO_DEMO_CAPTURE recovery states. DEMO and LIVE are DATA SOURCES that
  1020→  //   feed the SAME workflow — never implicit workspace states.
  1021→  const workflowStage: WorkflowStage = deriveWorkflowStage({
  1022→    hasLocation: !!selectedLocation,
  1023→    hasAoi: !!analysisAoi,
  1024→    aoiValid: aoiValidation.valid,
  1025→    ready: generateReadiness.enabled,
  1026→    loading,
  1027→    hasResult: !!jointDecision,
  1028→    errorCode: errorDetails?.code ?? null,
  1029→  });
  1030→
  1031→  // Candidates to display on the map + evaluate:
  1032→  //   FIXTURE + genuine capture → the three application-defined DEMO
  1033→  //             candidates (evaluated against the captured field — NOT
  1034→  //             captured sites). No capture → NO candidates (never synthetic,
  1035→  //             never Manhattan candidates for another location).
  1036→  //   LIVE    → the user's real candidate sites only.
  1037→  const displayCandidates: CandidateLocation[] = useMemo(() => {
  1038→    if (mode === 'FIXTURE') {
  1039→      return demoCaptureAvailable ? DEMO_CANDIDATE_SITES : [];
  1040→    }
  1041→    return candidateSites.sites.map((s) => ({
  1042→      locationId: s.locationId,
  1043→      name: s.name,
  1044→      location: s.location,
  1045→    }));
  1046→  }, [mode, demoCaptureAvailable, candidateSites.sites]);
  1047→
  1048→  // Resolution display (Section 2): DEMO shows the fixture's ACTUAL captured
  1049→  // granularity; LIVE shows the provider granularity that will be sent.
  1050→  const resolutionDisplay = mode === 'FIXTURE' ? FIXTURE_DISPLAY_GRANULARITY : prefs.analysisResolution;
  1051→
  1052→  // Display timezone: DEMO is UTC-anchored (the capture's request hour is a
  1053→  // UTC instant) — displaying it in the selected location's timezone would
  1054→  // MISLABEL the captured wall-clock. LIVE uses the location's timezone.
  1055→  const displayTimezone = mode === 'FIXTURE' ? FIXTURE_TIMEZONE : (selectedLocation?.timezone || 'UTC');
  1056→
  1057→  const handleSelectScenario = useCallback((scenarioId: string) => {
  1058→    setSelectedScenarioId(scenarioId);
  1059→    selectedScenarioIdRef.current = scenarioId;
  1060→    if (jointDecisionRef.current) {
  1061→      const sc = scenarioAnalysisRef.current?.scenarios?.find((s) => s.scenarioId === scenarioId);
  1062→      fetchExplanation(jointDecisionRef.current, sc);
  1063→    }
  1064→  }, [fetchExplanation]);
  1065→
  1066→  const handleRefreshExplanation = useCallback(() => {
  1067→    if (jointDecisionRef.current) {
  1068→      const activeScen = scenarioAnalysisRef.current?.scenarios?.find(
  1069→        (s) => s.scenarioId === selectedScenarioIdRef.current
  1070→      ) ?? scenarioAnalysisRef.current?.scenarios?.[0];
  1071→      fetchExplanation(jointDecisionRef.current, activeScen);
  1072→    }
  1073→  }, [fetchExplanation]);
  1074→
  1075→  const handleGenerate = useCallback(() => {
  1076→    setAddSiteMode(false);
  1077→    const m = modeRef.current;
  1078→    const loc = selectedLocationRef.current;
  1079→    runDecisionPipeline(
  1080→      loc,
  1081→      m === 'FIXTURE' ? FIXTURE_CAPTURE_REQUEST_AOI : createAoiFromSpan(aoiCenterRef.current, prefs.analysisAoiSpanMetres, prefs.analysisAreaShape),
  1082→      temporalInputRef.current,
  1083→      m === 'FIXTURE' ? FIXTURE_TIMEZONE : (loc?.timezone || 'UTC'),
  1084→      m,
  1085→      m === 'LIVE'
  1086→        ? candidateSitesRef.current.map((s) => ({
  1087→            locationId: s.locationId,
  1088→            name: s.name,
  1089→            location: s.location,
  1090→          }))
  1091→        : undefined
  1092→    );
  1093→  }, [runDecisionPipeline, prefs.analysisAoiSpanMetres, prefs.analysisAreaShape]);
  1094→
  1095→  const handleRetry = useCallback(() => {
  1096→    const m = modeRef.current;
  1097→    const loc = selectedLocationRef.current;
  1098→    runDecisionPipeline(
  1099→      loc,
  1100→      m === 'FIXTURE' ? FIXTURE_CAPTURE_REQUEST_AOI : createAoiFromSpan(aoiCenterRef.current, prefs.analysisAoiSpanMetres, prefs.analysisAreaShape),
  1101→      temporalInputRef.current,
  1102→      m === 'FIXTURE' ? FIXTURE_TIMEZONE : (loc?.timezone || 'UTC'),
  1103→      m,
  1104→      m === 'LIVE'
  1105→        ? candidateSitesRef.current.map((s) => ({
  1106→            locationId: s.locationId,
  1107→            name: s.name,
  1108→            location: s.location,
  1109→          }))
  1110→        : undefined
  1111→    );
  1112→  }, [runDecisionPipeline, prefs.analysisAoiSpanMetres, prefs.analysisAreaShape]);
  1113→
  1114→  // Alternative-location buttons share the FULL location-selection semantics:
  1115→  // DEMO + capture → load the captured dataset (free); DEMO without capture →
  1116→  // honest NO_DEMO_CAPTURE; LIVE → configure, Generate spends credits only
  1117→  // when explicitly pressed (credit safety).
  1118→  const handleSelectAltLocation = useCallback((loc: NamedLocation) => {
  1119→    handleSelectLocation(loc);
  1120→  }, [handleSelectLocation]);
  1121→
  1122→  // ───────────────────────────────────────────────────────────────────────────
  1123→  // RENDER
  1124→  // ───────────────────────────────────────────────────────────────────────────
  1125→
  1126→  return (
  1127→    <main
  1128→      className="min-h-screen flex flex-col bg-surface-bg text-text-primary"
  1129→      style={{ fontFamily: "'Inter', system-ui, sans-serif" }}
  1130→      data-workflow-stage={workflowStage}
  1131→    >
  1132→      <Header
  1133→        mode={mode}
  1134→        unit={unit}
  1135→        onToggleUnit={setUnit}
  1136→        theme={theme}
  1137→        onToggleTheme={toggleTheme}
  1138→        onOpenSettings={() => setSettingsOpen(true)}
  1139→        fortyGuardStatus={fgStatus}
  1140→        aiStatus={aiStatus}
  1141→        aiProvider={aiProvider}
  1142→      />
  1143→
  1144→      <div className="flex-1 max-w-[1600px] w-full mx-auto px-4 sm:px-6 py-5 lg:py-6">
  1145→        <div className="grid grid-cols-1 lg:grid-cols-12 gap-5 lg:gap-6">
  1146→          {/* LEFT CONTROL RAIL */}
  1147→          <div className="lg:col-span-4 xl:col-span-3">
  1148→            <ControlRail
  1149→              mode={mode}
  1150→              selectedLocation={selectedLocation}
  1151→              analysisCenter={selectedLocation ? aoiCenter : undefined}
  1152→              demoCaptureAvailable={demoCaptureAvailable}
  1153→              stateLevelSelection={stateLevelSelection}
  1154→              temporalInput={temporalInput}
  1155→              onTemporalChange={handleTemporalChange}
  1156→              onGenerate={handleGenerate}
  1157→              loading={loading}
  1158→              generateDisabled={!generateReadiness.enabled}
  1159→              generateDisabledReason={generateReadiness.reason}
  1160→              onReset={handleResetAnalysis}
  1161→              onClearLocation={handleClearLocation}
  1162→              onSelectLocation={handleSelectLocation}
  1163→              onSwitchToLive={() => handleModeChange('LIVE')}
  1164→              fortyGuardStatus={fgStatus}
  1165→              fortyGuardHealth={fgHealth}
  1166→              aiStatus={aiStatus}
  1167→              aiHealth={aiHealth}
  1168→              fieldReady={fieldReady}
  1169→              onTestFortyGuard={() => checkFortyGuardHealth(mode)}
  1170→              onTestAI={checkAIHealth}
  1171→              candidateSites={candidateSites.sites}
  1172→              onRemoveSite={handleRemoveSite}
  1173→              onRenameSite={handleRenameSite}
  1174→              onToggleAddSiteMode={() => setAddSiteMode((v) => !v)}
  1175→              addSiteMode={addSiteMode}
  1176→              onAddSiteFromSearch={handleAddSiteFromSearch}
  1177→              fixtureGranularity={FIXTURE_DISPLAY_GRANULARITY}
  1178→              activeStateFilter={regionDisplayName || selectedLocation?.state}
  1179→            />
  1180→          </div>
  1181→
  1182→          {/* MAIN CANVAS */}
  1183→          <div className="lg:col-span-8 xl:col-span-9 space-y-5">
  1184→            {/* Error banner (highest priority in main canvas when present) */}
  1185→            {errorDetails && (
  1186→              <ErrorBanner
  1187→                errorDetails={errorDetails}
  1188→                mode={mode}
  1189→                altLocations={altLocations}
  1190→                onRetry={handleRetry}
  1191→                onSwitchToDemo={() => handleModeChange('FIXTURE')}
  1192→                onSwitchToLive={() => handleModeChange('LIVE')}
  1193→                onSelectAltLocation={handleSelectAltLocation}
  1194→              />
  1195→            )}
  1196→
  1197→            {/* 1. FORTYGUARD THERMAL FIELD — the visual hero */}
  1198→            <ThermalMapCanvas
  1199→              stage={workflowStage}
  1200→              locationName={selectedLocation?.name ?? 'No location selected'}
  1201→              baseTimestamp={spatialFieldMeta?.baseTimestamp}
  1202→              thermalCellCount={thermalCellCount}
  1203→              resolution={selectedLocation ? resolutionDisplay : undefined}
  1204→              mode={mode}
  1205→              loading={loading}
  1206→              selectedLocation={selectedLocation ?? undefined}
  1207→              analysisCenter={selectedLocation ? aoiCenter : undefined}
  1208→              temporalInput={temporalInput}
  1209→              timezone={displayTimezone}
  1210→              rankedCandidates={spatialDecision?.rankedLocations.map((r) => ({
  1211→                locationId: r.locationId,
  1212→                name: r.name,
  1213→                location: r.location,
  1214→              }))}
  1215→              recommendedLocationId={spatialDecision?.recommendedLocation.locationId}
  1216→            >
  1217→              <ThermalMap
  1218→                location={selectedLocation ? { latitude: selectedLocation.latitude, longitude: selectedLocation.longitude } : null}
  1219→                locationState={selectedLocation?.state}
  1220→                locationName={selectedLocation?.name}
  1221→                locationDraggable={mode === 'LIVE'}
  1222→                onMoveOperatingLocation={handleMoveOperatingLocation}
  1223→                regionBoundary={regionBoundary}
  1224→                regionMask={regionMask}
  1225→                regionDisplayName={regionDisplayName}
  1226→                analysisAoi={analysisAoi}
  1227→                aoiInvalid={aoiInvalid}
  1228→                aoiInvalidMessage={aoiValidation.valid ? undefined : aoiValidation.message}
  1229→                captureAoiLabel={mode === 'FIXTURE' && demoCaptureAvailable ? fixtureCaptureSpanLabel() : undefined}
  1230→                spatialField={spatialField}
  1231→                selectedTileId={decision?.evidenceBundle.selectedTileId}
  1232→                candidates={displayCandidates}
  1233→                candidatesDraggable={mode === 'LIVE'}
  1234→                onMoveCandidate={handleMoveCandidate}
  1235→                onRemoveCandidate={handleRemoveSite}
  1236→                recommendedLocationId={spatialDecision?.recommendedLocation.locationId}
  1237→                unit={unit}
  1238→                layerVisibility={prefs.mapLayerVisibility}
  1239→                onToggleLayer={prefSetters.setMapLayerVisibility}
  1240→                areaShape={prefs.analysisAreaShape}
  1241→                showLocationMarker={!!selectedLocation}
  1242→                emptyMapMessage={
  1243→                  workflowStage === 'NO_DEMO_CAPTURE'
  1244→                    ? 'No DEMO capture available for this location — switch to LIVE or pick a Manhattan DEMO location'
  1245→                    : 'Select a location to begin the analysis'
  1246→                }
  1247→                addSiteMode={addSiteMode}
  1248→                onAddSiteAt={handleAddSiteAt}
  1249→                onExitAddSiteMode={() => setAddSiteMode(false)}
  1250→                cameraBehavior={cameraBehavior}
  1251→                cameraNonce={cameraNonce}
  1252→              />
  1253→            </ThermalMapCanvas>
  1254→
  1255→            {/* 2. RECOMMENDED OPERATION */}
  1256→            {jointDecision && !errorDetails && (
  1257→              <RecommendedOperation
  1258→                jointDecision={jointDecision}
  1259→                unit={unit}
  1260→                timezone={displayTimezone}
  1261→                mode={mode}
  1262→                temporalInput={temporalInput}
  1263→              />
  1264→            )}
  1265→
  1266→            {/* 3. TOP CANDIDATES */}
  1267→            {jointDecision && !errorDetails && (
  1268→              <TopCandidates
  1269→                jointDecision={jointDecision}
  1270→                unit={unit}
  1271→                timezone={displayTimezone}
  1272→              />
  1273→            )}
  1274→
  1275→            {/* 4. WHAT-IF */}
  1276→            {scenarioAnalysis && scenarioAnalysis.scenarios.length > 0 && !errorDetails && (
  1277→              <WhatIfPanel
  1278→                scenarioAnalysis={scenarioAnalysis}
  1279→                selectedScenarioId={selectedScenarioId}
  1280→                onSelectScenario={handleSelectScenario}
  1281→                unit={unit}
  1282→                timezone={displayTimezone}
  1283→              />
  1284→            )}
  1285→
  1286→            {/* 5. GROUNDED AI EXPLANATION (subdued — not the hero) */}
  1287→            {explanation && !errorDetails && (
  1288→              <GroundedExplanation
  1289→                explanation={explanation}
  1290→                unit={unit}
  1291→                timezone={displayTimezone}
  1292→                explaining={explaining}
  1293→                onRefresh={handleRefreshExplanation}
  1294→              />
  1295→            )}
  1296→
  1297→            {/* Empty state before first run (EMPTY / LOCATION_SELECTED stages) */}
  1298→            {!jointDecision && !loading && !errorDetails && (
  1299→              <div
  1300→                className="rounded-xl border border-border bg-surface-card px-5 py-12 text-center"
  1301→                data-testid={workflowStage === 'EMPTY' ? 'empty-workspace-card' : 'pre-generate-card'}
  1302→              >
  1303→                <div className="text-2xl mb-2">🌡</div>
  1304→                {workflowStage === 'EMPTY' ? (
  1305→                  <>
  1306→                    <p className="text-text-primary text-sm font-bold">Select a location to begin</p>
  1307→                    <p className="text-text-muted text-xs mt-1.5 leading-relaxed max-w-md mx-auto">
  1308→                      Search a city, street, or address — or pick a preset in the left rail. Locations with a genuine
  1309→                      captured DEMO dataset load it explicitly; every other location runs LIVE against FortyGuard.
  1310→                    </p>
  1311→                    <p className="text-[10px] font-mono text-text-dimmed mt-3 tracking-wide">
  1312→                      LOCATION → ANALYSIS AOI → THERMAL OBSERVATIONS → CANDIDATES → RECOMMENDATION
  1313→                    </p>
  1314→                  </>
  1315→                ) : (
  1316→                  <>
  1317→                    <p className="text-text-muted text-sm font-medium">Generate a thermal field to see the recommended operational plan.</p>
  1318→                    <p className="text-text-dimmed text-xs mt-1">
  1319→                      {mode === 'LIVE'
  1320→                        ? 'Place candidate sites (+ Site on the map), set the WHEN date/time, then click Generate.'
  1321→                        : 'This location has a genuine captured dataset — Generate replays it (no provider request).'}
  1322→                    </p>
  1323→                  </>
  1324→                )}
  1325→              </div>
  1326→            )}
  1327→          </div>
  1328→        </div>
  1329→      </div>
  1330→
  1331→      {/* SETTINGS DRAWER — provider capability + diagnostics */}
  1332→      <SettingsDrawer open={settingsOpen} onOpenChange={setSettingsOpen} capability={capability} />
  1333→    </main>
  1334→  );
  1335→}
