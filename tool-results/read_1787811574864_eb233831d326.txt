    85→async function safeJsonFetch<T = Record<string, unknown>>(
    86→  url: string,
    87→  init?: RequestInit
    88→): Promise<{ ok: boolean; data: T | null }> {
    89→  try {
    90→    const res = await fetch(url, init);
    91→    const text = await res.text();
    92→    if (!text || !text.trim()) return { ok: res.ok, data: null };
    93→    try {
    94→      return { ok: res.ok, data: JSON.parse(text) as T };
    95→    } catch {
    96→      return { ok: false, data: null };
    97→    }
    98→  } catch {
    99→    return { ok: false, data: null };
   100→  }
   101→}

   460→          longitude: c.location.longitude,

   467→        spatialDecision?: SpatialDecisionResult;

   471→        spatialFieldMetadata?: typeof spatialFieldMeta;
   472→        error?: ProductionErrorDetails;

   474→        method: 'POST',
   475→        headers: { 'content-type': 'application/json' },
   476→        body: JSON.stringify(body),
   477→      });
   478→
   479→      // Discard stale responses
   480→      if (requestId !== activeRequestIdRef.current) return;
   481→
   482→      if (!ok || !data || !data.success || data.error) {
   483→        setDecision(null);
   484→        setSpatialDecision(null);
   485→        setJointDecision(null);
   486→        setScenarioAnalysis(null);
   487→        setExplanation(null);
   488→        setSpatialField(null);
   489→        setSpatialFieldMeta(null);
   490→        if (data?.error) {
   491→          setErrorDetails(data.error);
   492→        }
   493→        if (dataSourceMode === 'LIVE') setFgStatus('ERROR');
   494→        return;
   495→      }
   496→
   497→      setDecision(data.decision ?? null);
   498→      setSpatialDecision(data.spatialDecision ?? null);
   499→      setJointDecision(data.jointDecision ?? null);
   500→      setScenarioAnalysis(data.scenarioAnalysis ?? null);
   501→      setSpatialField(data.spatialField ?? null);
   502→      setSpatialFieldMeta(data.spatialFieldMetadata ?? null);
   503→      setErrorDetails(null);
