   107→/** Extract a flat [lng, lat] list from a Polygon or MultiPolygon geometry. */
   108→function extractCoords(geometry: { type: string; coordinates: unknown }): [number, number][] {
   109→  const pts: [number, number][] = [];
   110→  if (geometry.type === 'Polygon') {
   111→    for (const ring of (geometry.coordinates as number[][][])) {
   112→      for (const [lng, lat] of ring) {
   113→        if (Number.isFinite(lng) && Number.isFinite(lat)) pts.push([lng, lat]);
   114→      }
   115→    }
   116→  } else if (geometry.type === 'MultiPolygon') {
   117→    for (const poly of (geometry.coordinates as number[][][][])) {
   118→      for (const ring of poly) {
   119→        for (const [lng, lat] of ring) {
   120→          if (Number.isFinite(lng) && Number.isFinite(lat)) pts.push([lng, lat]);
   121→        }
   122→      }
   123→    }
   124→  }
   125→  return pts;
   126→}

   128→/**
   129→ * Compute bounding box covering the AOI, thermal field, candidates, and the
   130→ * selected location (small geographic context around the AOI — Section 12).
   131→ */
   132→function computeAoiBounds(
   133→  analysisAoi: PolygonAOI | null,
   134→  spatialField: PolygonAOI | null,
   135→  candidateLocs: LocationPoint[],
   136→  targetLoc: LocationPoint,
   137→): [[number, number], [number, number]] | null {
   138→  const allPts: [number, number][] = [];
   139→
   140→  if (Number.isFinite(targetLoc.longitude) && Number.isFinite(targetLoc.latitude)) {
   141→    allPts.push([targetLoc.longitude, targetLoc.latitude]);
   142→  }
   143→
   144→  if (analysisAoi && analysisAoi.features.length > 0) {
   145→    for (const f of analysisAoi.features) {
   146→      const pts = extractCoords(f.geometry as { type: string; coordinates: unknown });
   147→      for (const [lng, lat] of pts) {
   148→        if (Math.abs(lng - targetLoc.longitude) < 0.35 && Math.abs(lat - targetLoc.latitude) < 0.35) {
   149→          allPts.push([lng, lat]);
   150→        }
   151→      }
   152→    }
   153→  }
   154→
   155→  if (spatialField && spatialField.features.length > 0) {
   156→    for (const f of spatialField.features) {
   157→      const pts = extractCoords(f.geometry as { type: string; coordinates: unknown });
   158→      for (const [lng, lat] of pts) {
   159→        if (Math.abs(lng - targetLoc.longitude) < 0.35 && Math.abs(lat - targetLoc.latitude) < 0.35) {
   160→          allPts.push([lng, lat]);
   161→        }
   162→      }
   163→    }
   164→  }
   165→
   166→  for (const { longitude, latitude } of candidateLocs) {
   167→    if (Number.isFinite(longitude) && Number.isFinite(latitude)) {
   168→      if (Math.abs(longitude - targetLoc.longitude) < 0.35 && Math.abs(latitude - targetLoc.latitude) < 0.35) {
   169→        allPts.push([longitude, latitude]);
   170→      }
   171→    }
   172→  }
   173→
   174→  if (allPts.length === 0) return null;
   175→
   176→  let minLng = allPts[0][0], maxLng = allPts[0][0];
   177→  let minLat = allPts[0][1], maxLat = allPts[0][1];
   178→  for (const [lng, lat] of allPts) {
   179→    if (lng < minLng) minLng = lng;
   180→    if (lng > maxLng) maxLng = lng;
   181→    if (lat < minLat) minLat = lat;
   182→    if (lat > maxLat) maxLat = lat;
   183→  }
   184→
   185→  const padRatio = 0.25;
   186→  const padLng = Math.max(0.001, (maxLng - minLng) * padRatio);
   187→  const padLat = Math.max(0.001, (maxLat - minLat) * padRatio);
   188→
   189→  return [
   190→    [minLng - padLng, minLat - padLat],
   191→    [maxLng + padLng, maxLat + padLat],
   192→  ];
   193→}

   195→/** Bounding box of the geographic region boundary (state-level fit). */
   196→function computeRegionBounds(
   197→  regionBoundary: PolygonAOI | null,
   198→): [[number, number], [number, number]] | null {
   199→  if (!regionBoundary || regionBoundary.features.length === 0) return null;
   200→  const allPts: [number, number][] = [];
   201→  for (const f of regionBoundary.features) {
   202→    allPts.push(...extractCoords(f.geometry as { type: string; coordinates: unknown }));
   203→  }
   204→  if (allPts.length === 0) return null;
   205→
   206→  let minLng = allPts[0][0], maxLng = allPts[0][0];
   207→  let minLat = allPts[0][1], maxLat = allPts[0][1];
   208→  for (const [lng, lat] of allPts) {
   209→    if (lng < minLng) minLng = lng;
   210→    if (lng > maxLng) maxLng = lng;
   211→    if (lat < minLat) minLat = lat;
   212→    if (lat > maxLat) maxLat = lat;
   213→  }
   214→  const padRatio = 0.08;
   215→  const padLng = Math.max(0.01, (maxLng - minLng) * padRatio);
   216→  const padLat = Math.max(0.01, (maxLat - minLat) * padRatio);
   217→  return [
   218→    [minLng - padLng, minLat - padLat],
   219→    [maxLng + padLng, maxLat + padLat],
   220→  ];
   221→}

   223→/** Return true if the spatial field contains at least one feature with a valid temperature. */
   224→function hasRenderableTemperatureData(aoi: PolygonAOI | null | undefined): boolean {
   225→  return !!aoi && aoi.features.length > 0 && aoi.features.some((f) =>
   226→    Number.isFinite(Number(f.properties?.average_temperature))
   227→  );
   228→}

   230→/**
   231→ * Drag-affordance anchor for the AOI handle: the NORTH-EAST point of the AOI
   232→ * boundary — the exact NE corner vertex for squares, the 45° boundary point
   233→ * for circles.
   234→ *
   235→ * The handle previously sat at the AOI CENTER, where the recommended-site
   236→ * marker (40px, z-index 30) fully covered the 34px handle (z-index 25)
   237→ * whenever the winner candidate is the analysis center, making the AOI
   238→ * undraggable. Anchoring the handle at the NE boundary keeps the affordance
   239→ * reachable without touching any candidate marker. Dragging translates the
   240→ * WHOLE AOI by the handle's movement delta — the AOI center is NOT moved to
   241→ * the handle position, and the canonical geometry (shape/size) is unchanged.
   242→ */
   243→function getAoiHandleAnchor(aoi: PolygonAOI): LocationPoint | null {
   244→  const feat = aoi.features[0];
   245→  const geom = feat?.geometry as { type: string; coordinates: number[][][] } | undefined;
   246→  if (!feat || !geom || !Array.isArray(geom.coordinates?.[0])) return null;
   247→  const ring = geom.coordinates[0];
   248→  if (ring.length < 3) return null;
   249→
   250→  const center = getAoiCenter(aoi);
   251→  if (!center) return null;
   252→
   253→  let maxLng = -Infinity, maxLat = -Infinity;
   254→  for (const [lng, lat] of ring) {
   255→    if (lng > maxLng) maxLng = lng;
   256→    if (lat > maxLat) maxLat = lat;
   257→  }
   258→
   259→  const props = (feat.properties ?? {}) as { shape?: string };
   260→  if (props.shape === 'circle') {
   261→    // 45° point ON the circle (NE), derived from the ring's bbox radii.
   262→    const rLng = maxLng - center.longitude;
   263→    const rLat = maxLat - center.latitude;
   264→    return {
   265→      longitude: center.longitude + rLng * Math.SQRT1_2,
   266→      latitude: center.latitude + rLat * Math.SQRT1_2,
   267→    };
   268→  }
   269→
   270→  // Square: the exact NE corner vertex of the ring.
   271→  return { longitude: maxLng, latitude: maxLat };
   272→}
