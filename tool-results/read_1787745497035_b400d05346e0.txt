    70→async function safeJsonFetch<T = Record<string, unknown>>(
    71→  url: string,
    72→  init?: RequestInit
    73→): Promise<{ ok: boolean; data: T | null }> {
    74→  try {
    75→    const res = await fetch(url, init);
    76→    const text = await res.text();
    77→    if (!text || !text.trim()) return { ok: res.ok, data: null };
    78→    try {
    79→      return { ok: res.ok, data: JSON.parse(text) as T };
    80→    } catch {
    81→      return { ok: false, data: null };
    82→    }
    83→  } catch {
    84→    return { ok: false, data: null };
    85→  }
    86→}

   527→
   528→  // ───────────────────────────────────────────────────────────────────────────
   529→  // Derived values
   530→  // ───────────────────────────────────────────────────────────────────────────
   531→
   532→  const isFixtureMismatch = mode === 'FIXTURE' && !isLocationCoveredByFixture(selectedLocation);
   533→  const activeScenario = scenarioAnalysis?.scenarios?.find((s) => s.scenarioId === selectedScenarioId) ?? scenarioAnalysis?.scenarios?.[0];
   534→  const fieldReady = !!spatialField && !loading;
   535→  const altLocations = METROPOLITAN_LOCATIONS.filter((l) => !l.isDemoOnly).slice(0, 4);
   536→  const thermalCellCount = spatialField?.features?.length ?? 0;
   537→  const aiProvider = aiHealth?.provider;
   538→
   539→  const handleSelectScenario = useCallback((scenarioId: string) => {
   540→    setSelectedScenarioId(scenarioId);
   541→    selectedScenarioIdRef.current = scenarioId;
   542→    if (jointDecisionRef.current) {
   543→      const sc = scenarioAnalysisRef.current?.scenarios?.find((s) => s.scenarioId === scenarioId);
   544→      fetchExplanation(jointDecisionRef.current, sc);
   545→    }
   546→  }, [fetchExplanation]);
   547→
   548→  const handleRefreshExplanation = useCallback(() => {
   549→    if (jointDecisionRef.current) {
   550→      const activeScen = scenarioAnalysisRef.current?.scenarios?.find(
   551→        (s) => s.scenarioId === selectedScenarioIdRef.current
   552→      ) ?? scenarioAnalysisRef.current?.scenarios?.[0];
   553→      fetchExplanation(jointDecisionRef.current, activeScen);
   554→    }
   555→  }, [fetchExplanation]);
   556→
   557→  const handleGenerate = useCallback(() => {
   558→    runDecisionPipeline(
   559→      selectedLocationRef.current,
   560→      temporalInputRef.current,
   561→      selectedLocationRef.current.timezone || 'UTC',
   562→      modeRef.current
   563→    );
   564→  }, [runDecisionPipeline]);
   565→
   566→  const handleRetry = useCallback(() => {
   567→    runDecisionPipeline(
   568→      selectedLocationRef.current,
   569→      temporalInputRef.current,
   570→      selectedLocationRef.current.timezone || 'UTC',
   571→      modeRef.current
   572→    );
   573→  }, [runDecisionPipeline]);
   574→
   575→  // Credit safety: selecting an alternative location NEVER triggers a provider
   576→  // request by itself. It updates the selection + WHEN defaults and clears the
   577→  // previous results; the user explicitly presses Generate to spend credits.
   578→  const handleSelectAltLocation = useCallback((loc: NamedLocation) => {
   579→    setSelectedLocation(loc);
   580→    selectedLocationRef.current = loc;
   581→    if (modeRef.current === 'FIXTURE' && isLocationCoveredByFixture(loc)) {
   582→      const fixtureTemporal = buildFixtureTemporalInput();
   583→      setTemporalInput(fixtureTemporal);
   584→      temporalInputRef.current = fixtureTemporal;
   585→    } else {
   586→      const liveDefault = defaultTemporalInput(loc.timezone);
   587→      setTemporalInput(liveDefault);
   588→      temporalInputRef.current = liveDefault;
   589→    }
   590→    setDecision(null);
   591→    setSpatialDecision(null);
   592→    setJointDecision(null);
   593→    setScenarioAnalysis(null);
   594→    setExplanation(null);
   595→    setSpatialField(null);
   596→    setSpatialFieldMeta(null);
   597→    setErrorDetails(null);
   598→  }, []);
   599→
   600→  // ───────────────────────────────────────────────────────────────────────────
   601→  // RENDER
   602→  // ───────────────────────────────────────────────────────────────────────────
   603→
   604→  return (
   605→    <main className="min-h-screen flex flex-col bg-surface-bg text-text-primary" style={{ fontFamily: "'Inter', system-ui, sans-serif" }}>
   606→      <Header
   607→        mode={mode}
   608→        unit={unit}
   609→        onToggleUnit={setUnit}
   610→        theme={theme}
   611→        onToggleTheme={toggleTheme}
   612→        onOpenSettings={() => setSettingsOpen(true)}
   613→        fortyGuardStatus={fgStatus}
   614→        aiStatus={aiStatus}
   615→        aiProvider={aiProvider}
   616→      />
   617→
   618→      <div className="flex-1 max-w-[1600px] w-full mx-auto px-4 sm:px-6 py-5 lg:py-6">
   619→        <div className="grid grid-cols-1 lg:grid-cols-12 gap-5 lg:gap-6">
   620→          {/* LEFT CONTROL RAIL */}
   621→          <div className="lg:col-span-4 xl:col-span-3">
   622→            <ControlRail
   623→              mode={mode}
   624→              selectedLocation={selectedLocation}
   625→              temporalInput={temporalInput}
   626→              onTemporalChange={handleTemporalChange}
   627→              onGenerate={handleGenerate}
   628→              loading={loading}
   629→              onSelectLocation={handleSelectLocation}
   630→              onSwitchToLive={() => handleModeChange('LIVE')}
   631→              fortyGuardStatus={fgStatus}
   632→              fortyGuardHealth={fgHealth}
   633→              aiStatus={aiStatus}
   634→              aiHealth={aiHealth}
   635→              fieldReady={fieldReady}
   636→              onTestFortyGuard={() => checkFortyGuardHealth(mode)}
   637→              onTestAI={checkAIHealth}
   638→            />
   639→          </div>
   640→
   641→          {/* MAIN CANVAS */}
   642→          <div className="lg:col-span-8 xl:col-span-9 space-y-5">
   643→            {/* Error banner (highest priority in main canvas when present) */}
   644→            {errorDetails && (
   645→              <ErrorBanner
   646→                errorDetails={errorDetails}
   647→                mode={mode}
   648→                altLocations={altLocations}
   649→                onRetry={handleRetry}
   650→                onSwitchToDemo={() => handleModeChange('FIXTURE')}
   651→                onSelectAltLocation={handleSelectAltLocation}
   652→              />
   653→            )}
   654→
   655→            {/* 1. FORTYGUARD THERMAL FIELD — the visual hero */}
   656→            <ThermalMapCanvas
   657→              locationName={selectedLocation.name}
   658→              baseTimestamp={spatialFieldMeta?.baseTimestamp}
   659→              thermalCellCount={thermalCellCount}
   660→              resolution={prefs.analysisResolution}
   661→              mode={mode}
   662→              loading={loading}
   663→              selectedLocation={selectedLocation}
   664→              temporalInput={temporalInput}
   665→              timezone={selectedLocation.timezone}
   666→              rankedCandidates={spatialDecision?.rankedLocations.map((r) => ({
   667→                locationId: r.locationId,
   668→                name: r.name,
   669→                location: r.location,
   670→              }))}
   671→              recommendedLocationId={spatialDecision?.recommendedLocation.locationId}
   672→            >
   673→              <ThermalMap
   674→                location={{ latitude: selectedLocation.latitude, longitude: selectedLocation.longitude }}
   675→                locationState={selectedLocation.state}
   676→                locationName={selectedLocation.name}
   677→                regionBoundary={regionBoundary}
   678→                regionMask={regionMask}
   679→                analysisAoi={analysisAoi}
   680→                spatialField={spatialField}
   681→                selectedTileId={decision?.evidenceBundle.selectedTileId}
   682→                candidates={spatialDecision?.rankedLocations.map((r) => ({
   683→                  locationId: r.locationId,
   684→                  name: r.name,
   685→                  location: r.location,
   686→                }))}
   687→                recommendedLocationId={spatialDecision?.recommendedLocation.locationId}
   688→                unit={unit}
   689→                layerVisibility={prefs.mapLayerVisibility}
   690→                onToggleLayer={prefSetters.setMapLayerVisibility}
   691→                areaShape={prefs.analysisAreaShape}
   692→                aoiHalfSideMetres={prefs.analysisAoiHalfSideMetres}
   693→                onChangeAreaShape={prefSetters.setAnalysisAreaShape}
   694→                onChangeAoiHalfSideMetres={prefSetters.setAnalysisAoiHalfSideMetres}
   695→              />
   696→            </ThermalMapCanvas>
   697→
   698→            {/* 2. RECOMMENDED OPERATION */}
   699→            {jointDecision && !errorDetails && (
   700→              <RecommendedOperation
   701→                jointDecision={jointDecision}
   702→                unit={unit}
   703→                timezone={selectedLocation.timezone}
   704→                mode={mode}
   705→                temporalInput={temporalInput}
   706→              />
   707→            )}
   708→
   709→            {/* 3. TOP CANDIDATES */}
   710→            {jointDecision && !errorDetails && (
   711→              <TopCandidates
   712→                jointDecision={jointDecision}
   713→                unit={unit}
   714→                timezone={selectedLocation.timezone}
   715→              />
   716→            )}
   717→
   718→            {/* 4. WHAT-IF */}
   719→            {scenarioAnalysis && scenarioAnalysis.scenarios.length > 0 && !errorDetails && (
   720→              <WhatIfPanel
   721→                scenarioAnalysis={scenarioAnalysis}
   722→                selectedScenarioId={selectedScenarioId}
   723→                onSelectScenario={handleSelectScenario}
   724→                unit={unit}
   725→                timezone={selectedLocation.timezone}
   726→              />
   727→            )}
   728→
   729→            {/* 5. GROUNDED AI EXPLANATION (subdued — not the hero) */}
   730→            {explanation && !errorDetails && (
   731→              <GroundedExplanation
   732→                explanation={explanation}
   733→                unit={unit}
   734→                timezone={selectedLocation.timezone}
   735→                explaining={explaining}
   736→                onRefresh={handleRefreshExplanation}
   737→              />
   738→            )}
   739→
   740→            {/* Empty state before first run */}
   741→            {!jointDecision && !loading && !errorDetails && (
   742→              <div className="rounded-xl border border-border bg-surface-card px-5 py-12 text-center">
   743→                <div className="text-2xl mb-2">🌡</div>
   744→                <p className="text-text-muted text-sm font-medium">Generate a thermal field to see the recommended operational plan.</p>
   745→                <p className="text-text-dimmed text-xs mt-1">Select a location, set the WHEN date/time, then click Generate.</p>
   746→              </div>
   747→            )}
   748→          </div>
   749→        </div>
   750→      </div>
   751→
   752→      {/* SETTINGS DRAWER — now surfaces provider capability (Section 1) */}
   753→      <SettingsDrawer open={settingsOpen} onOpenChange={setSettingsOpen} capability={capability} />
   754→    </main>
   755→  );
   756→}
