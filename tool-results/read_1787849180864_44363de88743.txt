   108→/** Extract a flat [lng, lat] list from a Polygon or MultiPolygon geometry. */
   109→function extractCoords(geometry: { type: string; coordinates: unknown }): [number, number][] {
   110→  const pts: [number, number][] = [];
   111→  if (geometry.type === 'Polygon') {
   112→    for (const ring of (geometry.coordinates as number[][][])) {
   113→      for (const [lng, lat] of ring) {
   114→        if (Number.isFinite(lng) && Number.isFinite(lat)) pts.push([lng, lat]);
   115→      }
   116→    }
   117→  } else if (geometry.type === 'MultiPolygon') {
   118→    for (const poly of (geometry.coordinates as number[][][][])) {
   119→      for (const ring of poly) {
   120→        for (const [lng, lat] of ring) {
   121→          if (Number.isFinite(lng) && Number.isFinite(lat)) pts.push([lng, lat]);
   122→        }
   123→      }
   124→    }
   125→  }
   126→  return pts;
   127→}

   129→/** Compute bounding box covering the AOI, thermal field, candidates, and operating location. */
   130→function computeAoiBounds(
   131→  analysisAoi: PolygonAOI | null,
   132→  spatialField: PolygonAOI | null,
   133→  candidateLocs: LocationPoint[],
   134→  targetLoc: LocationPoint,
   135→): [[number, number], [number, number]] | null {
   136→  const allPts: [number, number][] = [];
   137→
   138→  if (Number.isFinite(targetLoc.longitude) && Number.isFinite(targetLoc.latitude)) {
   139→    allPts.push([targetLoc.longitude, targetLoc.latitude]);
   140→  }
   141→
   142→  if (analysisAoi && analysisAoi.features.length > 0) {
   143→    for (const f of analysisAoi.features) {
   144→      const pts = extractCoords(f.geometry as { type: string; coordinates: unknown });
   145→      for (const [lng, lat] of pts) {
   146→        if (Math.abs(lng - targetLoc.longitude) < 0.35 && Math.abs(lat - targetLoc.latitude) < 0.35) {
   147→          allPts.push([lng, lat]);
   148→        }
   149→      }
   150→    }
   151→  }
   152→
   153→  if (spatialField && spatialField.features.length > 0) {
   154→    for (const f of spatialField.features) {
   155→      const pts = extractCoords(f.geometry as { type: string; coordinates: unknown });
   156→      for (const [lng, lat] of pts) {
   157→        if (Math.abs(lng - targetLoc.longitude) < 0.35 && Math.abs(lat - targetLoc.latitude) < 0.35) {
   158→          allPts.push([lng, lat]);
   159→        }
   160→      }
   161→    }
   162→  }
   163→
   164→  for (const { longitude, latitude } of candidateLocs) {
   165→    if (Number.isFinite(longitude) && Number.isFinite(latitude)) {
   166→      if (Math.abs(longitude - targetLoc.longitude) < 0.35 && Math.abs(latitude - targetLoc.latitude) < 0.35) {
   167→        allPts.push([longitude, latitude]);
   168→      }
   169→    }
   170→  }
   171→
   172→  if (allPts.length === 0) return null;
   173→
   174→  let minLng = allPts[0][0], maxLng = allPts[0][0];
   175→  let minLat = allPts[0][1], maxLat = allPts[0][1];
   176→  for (const [lng, lat] of allPts) {
   177→    if (lng < minLng) minLng = lng;
   178→    if (lng > maxLng) maxLng = lng;
   179→    if (lat < minLat) minLat = lat;
   180→    if (lat > maxLat) maxLat = lat;
   181→  }
   182→
   183→  const padRatio = 0.25;
   184→  const padLng = Math.max(0.001, (maxLng - minLng) * padRatio);
   185→  const padLat = Math.max(0.001, (maxLat - minLat) * padRatio);
   186→
   187→  return [
   188→    [minLng - padLng, minLat - padLat],
   189→    [maxLng + padLng, maxLat + padLat],
   190→  ];
   191→}

   193→/** Bounding box of the geographic region boundary. */
   194→function computeRegionBounds(
   195→  regionBoundary: PolygonAOI | null,
   196→): [[number, number], [number, number]] | null {
   197→  if (!regionBoundary || regionBoundary.features.length === 0) return null;
   198→  const allPts: [number, number][] = [];
   199→  for (const f of regionBoundary.features) {
   200→    allPts.push(...extractCoords(f.geometry as { type: string; coordinates: unknown }));
   201→  }
   202→  if (allPts.length === 0) return null;
   203→
   204→  let minLng = allPts[0][0], maxLng = allPts[0][0];
   205→  let minLat = allPts[0][1], maxLat = allPts[0][1];
   206→  for (const [lng, lat] of allPts) {
   207→    if (lng < minLng) minLng = lng;
   208→    if (lng > maxLng) maxLng = lng;
   209→    if (lat < minLat) minLat = lat;
   210→    if (lat > maxLat) maxLat = lat;
   211→  }
   212→  const padRatio = 0.08;
   213→  const padLng = Math.max(0.01, (maxLng - minLng) * padRatio);
   214→  const padLat = Math.max(0.01, (maxLat - minLat) * padRatio);
   215→  return [
   216→    [minLng - padLng, minLat - padLat],
   217→    [maxLng + padLng, maxLat + padLat],
   218→  ];
   219→}

   221→/** Return true if the spatial field contains at least one feature with a valid temperature. */
   222→function hasRenderableTemperatureData(aoi: PolygonAOI | null | undefined): boolean {
   223→  return (
   224→    !!aoi &&
   225→    aoi.features.length > 0 &&
   226→    aoi.features.some((f) => {
   227→      const p = f.properties;
   228→      const v = p?.average_temperature ?? p?.temperature ?? p?.temp ?? p?.value;
   229→      return Number.isFinite(Number(v));
   230→    })
   231→  );
   232→}

   234→/** Apply valid/invalid paint to the canonical AOI layers. */
   235→function applyAoiValidityPaint(map: MapLibreMap, invalid: boolean, isDark: boolean): void {
   236→  try {
   237→    if (map.getLayer('aoi-outline')) {
   238→      map.setPaintProperty('aoi-outline', 'line-color', invalid ? '#ef4444' : (isDark ? '#fb7185' : '#be123c'));
   239→      map.setPaintProperty('aoi-outline', 'line-width', invalid ? 3.5 : 2.0);
   240→    }
   241→    if (map.getLayer('aoi-fill')) {
   242→      map.setPaintProperty('aoi-fill', 'fill-color', invalid ? '#ef4444' : '#f43f5e');
   243→      map.setPaintProperty('aoi-fill', 'fill-opacity', invalid ? 0.08 : 0.02);
   244→    }
   245→  } catch {
   246→    /* safe */
   247→  }
   248→}

   765→/** Visual edge fill helper so thermal tiles visually reach the AOI boundary with 0 gap */
   766→function alignThermalCellsToAoi(spatialField: PolygonAOI, aoi: PolygonAOI | null | undefined): PolygonAOI {
   767→  if (!spatialField || !spatialField.features || spatialField.features.length === 0) return spatialField;
   768→  if (!aoi || !aoi.features || aoi.features.length === 0) return spatialField;
   769→
   770→  const aoiRing = (aoi.features[0]?.geometry as { coordinates?: number[][][] })?.coordinates?.[0];
   771→  if (!aoiRing || aoiRing.length === 0) return spatialField;
   772→
   773→  let aoiMinLng = Infinity, aoiMaxLng = -Infinity, aoiMinLat = Infinity, aoiMaxLat = -Infinity;
   774→  for (const [lng, lat] of aoiRing) {
   775→    if (lng < aoiMinLng) aoiMinLng = lng;
   776→    if (lng > aoiMaxLng) aoiMaxLng = lng;
   777→    if (lat < aoiMinLat) aoiMinLat = lat;
   778→    if (lat > aoiMaxLat) aoiMaxLat = lat;
   779→  }
   780→
   781→  let cellMinLng = Infinity, cellMaxLng = -Infinity, cellMinLat = Infinity, cellMaxLat = -Infinity;
   782→  for (const f of spatialField.features) {
   783→    const ring = (f.geometry as { coordinates?: number[][][] })?.coordinates?.[0];
   784→    if (ring) {
   785→      for (const [lng, lat] of ring) {
   786→        if (lng < cellMinLng) cellMinLng = lng;
   787→        if (lng > cellMaxLng) cellMaxLng = lng;
   788→        if (lat < cellMinLat) cellMinLat = lat;
   789→        if (lat > cellMaxLat) cellMaxLat = lat;
   790→      }
   791→    }
   792→  }
   793→
   794→  const epsLng = Math.max(0.0001, (cellMaxLng - cellMinLng) / 35);
   795→  const epsLat = Math.max(0.0001, (cellMaxLat - cellMinLat) / 35);
   796→
   797→  return {
   798→    type: 'FeatureCollection',
   799→    features: spatialField.features.map((f) => {
   800→      const geom = f.geometry as { type: 'Polygon' | 'MultiPolygon'; coordinates: number[][][] };
   801→      if (!geom || !geom.coordinates || !geom.coordinates[0]) return f;
   802→      const newRing = geom.coordinates[0].map(([lng, lat]) => {
   803→        let adjLng = lng;
   804→        let adjLat = lat;
   805→        if (Math.abs(lng - cellMinLng) <= epsLng) adjLng = aoiMinLng;
   806→        else if (Math.abs(lng - cellMaxLng) <= epsLng) adjLng = aoiMaxLng;
   807→        if (Math.abs(lat - cellMinLat) <= epsLat) adjLat = aoiMinLat;
   808→        else if (Math.abs(lat - cellMaxLat) <= epsLat) adjLat = aoiMaxLat;
   809→        return [adjLng, adjLat];
   810→      });
   811→      return {
   812→        ...f,
   813→        geometry: {
   814→          ...geom,
   815→          coordinates: [newRing],
   816→        },
   817→      };
   818→    }),
   819→  };
   820→}
