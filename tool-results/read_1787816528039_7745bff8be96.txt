    88→async function safeJsonFetch<T = Record<string, unknown>>(
    89→  url: string,
    90→  init?: RequestInit
    91→): Promise<{ ok: boolean; data: T | null }> {
    92→  try {
    93→    const res = await fetch(url, init);
    94→    const text = await res.text();
    95→    if (!text || !text.trim()) return { ok: res.ok, data: null };
    96→    try {
    97→      return { ok: res.ok, data: JSON.parse(text) as T };
    98→    } catch {
    99→      return { ok: false, data: null };
   100→    }
   101→  } catch {
   102→    return { ok: false, data: null };
   103→  }
   104→}

   343→  const runDecisionPipeline = useCallback(async (

   395→        category: 'VALIDATION',
   396→      });
   397→      return;
   398→    }
   399→
   400→    // Validate the explicit temporal input before sending (Section 14).

   402→      setLoading(false);

   410→    }

   414→      setLoading(false);
   415→      setErrorDetails({
   416→        code: 'CANDIDATES_REQUIRED',
   417→        message: 'No candidate sites provided. LIVE mode never fabricates candidate sites — add at least one candidate site inside the analysis area.',

   419→        category: 'VALIDATION',
   420→      });
   421→      return;
   422→    }
   423→
   424→    // LIVE mode: every candidate must lie INSIDE the canonical AOI (Section 9).
   425→    if (dataSourceMode === 'LIVE' && candidates) {
   426→      const outside = candidates.find((c) => !isPointInAoi(c.location, aoi));
   427→      if (outside) {
   428→        setLoading(false);
   429→        setErrorDetails({
   430→          code: 'CANDIDATE_OUTSIDE_AOI',

   433→          category: 'VALIDATION',
   434→        });
   435→        return;
   436→      }
   437→    }
   438→
   439→    // DEMO capture-extent gate (P2): DEMO replays ONE fixed captured field.
   440→    // If the AOI was dragged outside the captured extent, no provider data can

   443→      setLoading(false);
   444→      setErrorDetails({
   445→        code: 'AOI_OUTSIDE_DEMO_CAPTURE',

   448→        category: 'COVERAGE',
   449→      });
   450→      return;
   451→    }
   452→
   453→    // DEMO canonical AOI: the CAPTURED REQUEST AREA (the genuine polygon_aoi
   454→    // the capture was requested with — mirrored from the fixture metadata).
   455→    // The captured field is replayed verbatim inside it; DEMO never submits a
   456→    // client-invented geometry for the capture. (LIVE submits the user's
   457→    // canonical AOI exactly as rendered — including any drag.)
   458→    const requestAoi = dataSourceMode === 'FIXTURE' ? FIXTURE_CAPTURE_REQUEST_AOI : aoi;

   460→    try {
   461→      const body: Record<string, unknown> = {
   462→        latitude: loc.latitude,
   463→        longitude: loc.longitude,
   464→        durationHours: deriveDurationHours(temporal),
   465→        mode: dataSourceMode,
   466→        // DEMO sends the capture's ACTUAL granularity (100m); LIVE sends the
   467→        // user-selected provider granularity.
   468→        granularity: dataSourceMode === 'FIXTURE' ? FIXTURE_DISPLAY_GRANULARITY : prefs.analysisResolution,
   469→        analysisAreaShape: prefs.analysisAreaShape,
   470→        // Canonical analysis AOI — the EXACT geometry rendered on the map
   471→        // (including any drag the user performed).
   472→        analysisAoi: requestAoi,
   473→        temporalInput: temporal,
   474→        timezone: tz,
   475→      };
   476→
   477→      // FIXTURE mode sends NO candidates — the server uses the actual three
   478→      // captured Manhattan sites. LIVE sends the user's real candidates.
   479→      if (dataSourceMode === 'LIVE' && candidates && candidates.length > 0) {
   480→        body.candidates = candidates.map((c) => ({
   481→          locationId: c.locationId,
   482→          name: c.name,
   483→          latitude: c.location.latitude,
   484→          longitude: c.location.longitude,
   485→        }));
   486→      }
   487→
   488→      const { ok, data } = await safeJsonFetch<{
   489→        success: boolean;
   490→        decision?: DecisionResult;
   491→        spatialDecision?: SpatialDecisionResult;
   492→        jointDecision?: JointDecisionResult;
   493→        scenarioAnalysis?: ScenarioAnalysisResult;
   494→        spatialField?: PolygonAOI | null;
   495→        spatialFieldMetadata?: typeof spatialFieldMeta;

   991→    aoiValid: aoiValidation.valid,
   992→    ready: generateReadiness.enabled,
   993→    loading,
   994→    hasResult: !!jointDecision,
   995→    errorCode: errorDetails?.code ?? null,
   996→  });
   997→
   998→  // Candidates to display on the map + evaluate:
   999→  //   FIXTURE + genuine capture → the three application-defined DEMO
  1000→  //             candidates (evaluated against the captured field — NOT
  1001→  //             captured sites). No capture → NO candidates (never synthetic,
  1002→  //             never Manhattan candidates for another location).
  1003→  //   LIVE    → the user's real candidate sites only.
  1004→  const displayCandidates: CandidateLocation[] = mode === 'FIXTURE'
  1005→    ? (demoCaptureAvailable ? DEMO_CANDIDATE_SITES : [])
  1006→    : candidateSites.sites.map((s) => ({
  1007→        locationId: s.locationId,
  1008→        name: s.name,
  1009→        location: s.location,
  1010→      }));
  1011→
  1012→  // Resolution display (Section 2): DEMO shows the fixture's ACTUAL captured
  1013→  // granularity; LIVE shows the provider granularity that will be sent.
  1014→  const resolutionDisplay = mode === 'FIXTURE' ? FIXTURE_DISPLAY_GRANULARITY : prefs.analysisResolution;
  1015→
  1016→  // Display timezone: DEMO is UTC-anchored (the capture's request hour is a
  1017→  // UTC instant) — displaying it in the selected location's timezone would
  1018→  // MISLABEL the captured wall-clock. LIVE uses the location's timezone.
  1019→  const displayTimezone = mode === 'FIXTURE' ? FIXTURE_TIMEZONE : (selectedLocation?.timezone || 'UTC');
  1020→
  1021→  const handleSelectScenario = useCallback((scenarioId: string) => {
  1022→    setSelectedScenarioId(scenarioId);
  1023→    selectedScenarioIdRef.current = scenarioId;
  1024→    if (jointDecisionRef.current) {
  1025→      const sc = scenarioAnalysisRef.current?.scenarios?.find((s) => s.scenarioId === scenarioId);
  1026→      fetchExplanation(jointDecisionRef.current, sc);
  1027→    }
  1028→  }, [fetchExplanation]);
  1029→
  1030→  const handleRefreshExplanation = useCallback(() => {
  1031→    if (jointDecisionRef.current) {
  1032→      const activeScen = scenarioAnalysisRef.current?.scenarios?.find(
  1033→        (s) => s.scenarioId === selectedScenarioIdRef.current
  1034→      ) ?? scenarioAnalysisRef.current?.scenarios?.[0];
  1035→      fetchExplanation(jointDecisionRef.current, activeScen);
  1036→    }
  1037→  }, [fetchExplanation]);
  1038→
  1039→  const handleGenerate = useCallback(() => {
  1040→    setAddSiteMode(false);
  1041→    const m = modeRef.current;
  1042→    const loc = selectedLocationRef.current;
  1043→    runDecisionPipeline(
  1044→      loc,
  1045→      m === 'FIXTURE' ? FIXTURE_CAPTURE_REQUEST_AOI : createAoiFromSpan(aoiCenterRef.current, prefs.analysisAoiSpanMetres, prefs.analysisAreaShape),
  1046→      temporalInputRef.current,
  1047→      m === 'FIXTURE' ? FIXTURE_TIMEZONE : (loc?.timezone || 'UTC'),
  1048→      m,
  1049→      m === 'LIVE'
  1050→        ? candidateSitesRef.current.map((s) => ({
  1051→            locationId: s.locationId,
  1052→            name: s.name,
  1053→            location: s.location,
  1054→          }))
  1055→        : undefined
  1056→    );
  1057→  }, [runDecisionPipeline, prefs.analysisAoiSpanMetres, prefs.analysisAreaShape]);
  1058→
  1059→  const handleRetry = useCallback(() => {
  1060→    const m = modeRef.current;
  1061→    const loc = selectedLocationRef.current;
  1062→    runDecisionPipeline(
  1063→      loc,
  1064→      m === 'FIXTURE' ? FIXTURE_CAPTURE_REQUEST_AOI : createAoiFromSpan(aoiCenterRef.current, prefs.analysisAoiSpanMetres, prefs.analysisAreaShape),
  1065→      temporalInputRef.current,
  1066→      m === 'FIXTURE' ? FIXTURE_TIMEZONE : (loc?.timezone || 'UTC'),
  1067→      m,
  1068→      m === 'LIVE'
  1069→        ? candidateSitesRef.current.map((s) => ({
  1070→            locationId: s.locationId,
  1071→            name: s.name,
  1072→            location: s.location,
  1073→          }))
  1074→        : undefined
  1075→    );
  1076→  }, [runDecisionPipeline, prefs.analysisAoiSpanMetres, prefs.analysisAreaShape]);
  1077→
  1078→  // Alternative-location buttons share the FULL location-selection semantics:
  1079→  // DEMO + capture → load the captured dataset (free); DEMO without capture →
  1080→  // honest NO_DEMO_CAPTURE; LIVE → configure, Generate spends credits only
  1081→  // when explicitly pressed (credit safety).
  1082→  const handleSelectAltLocation = useCallback((loc: NamedLocation) => {
  1083→    handleSelectLocation(loc);
  1084→  }, [handleSelectLocation]);
  1085→
  1086→  // ───────────────────────────────────────────────────────────────────────────
  1087→  // RENDER
  1088→  // ───────────────────────────────────────────────────────────────────────────
  1089→
  1090→  return (
  1091→    <main
  1092→      className="min-h-screen flex flex-col bg-surface-bg text-text-primary"
  1093→      style={{ fontFamily: "'Inter', system-ui, sans-serif" }}
  1094→      data-workflow-stage={workflowStage}
  1095→    >
  1096→      <Header
  1097→        mode={mode}
  1098→        unit={unit}
  1099→        onToggleUnit={setUnit}
  1100→        theme={theme}
  1101→        onToggleTheme={toggleTheme}
  1102→        onOpenSettings={() => setSettingsOpen(true)}
  1103→        fortyGuardStatus={fgStatus}
  1104→        aiStatus={aiStatus}
  1105→        aiProvider={aiProvider}
  1106→      />
  1107→
  1108→      <div className="flex-1 max-w-[1600px] w-full mx-auto px-4 sm:px-6 py-5 lg:py-6">
  1109→        <div className="grid grid-cols-1 lg:grid-cols-12 gap-5 lg:gap-6">
  1110→          {/* LEFT CONTROL RAIL */}
  1111→          <div className="lg:col-span-4 xl:col-span-3">
  1112→            <ControlRail
  1113→              mode={mode}
  1114→              selectedLocation={selectedLocation}
  1115→              analysisCenter={selectedLocation ? aoiCenter : undefined}
  1116→              demoCaptureAvailable={demoCaptureAvailable}
  1117→              stateLevelSelection={stateLevelSelection}
  1118→              temporalInput={temporalInput}
  1119→              onTemporalChange={handleTemporalChange}
  1120→              onGenerate={handleGenerate}
  1121→              loading={loading}
  1122→              generateDisabled={!generateReadiness.enabled}
  1123→              generateDisabledReason={generateReadiness.reason}
  1124→              onReset={handleResetAnalysis}
  1125→              onClearLocation={handleClearLocation}
  1126→              onSelectLocation={handleSelectLocation}
  1127→              onSwitchToLive={() => handleModeChange('LIVE')}
  1128→              fortyGuardStatus={fgStatus}
  1129→              fortyGuardHealth={fgHealth}
  1130→              aiStatus={aiStatus}
  1131→              aiHealth={aiHealth}
  1132→              fieldReady={fieldReady}
  1133→              onTestFortyGuard={() => checkFortyGuardHealth(mode)}
  1134→              onTestAI={checkAIHealth}
  1135→              candidateSites={candidateSites.sites}
  1136→              onRemoveSite={handleRemoveSite}
  1137→              onRenameSite={handleRenameSite}
  1138→              onToggleAddSiteMode={() => setAddSiteMode((v) => !v)}
  1139→              addSiteMode={addSiteMode}
  1140→              onAddSiteFromSearch={handleAddSiteFromSearch}
  1141→              fixtureGranularity={FIXTURE_DISPLAY_GRANULARITY}
  1142→            />
  1143→          </div>
  1144→
  1145→          {/* MAIN CANVAS */}
  1146→          <div className="lg:col-span-8 xl:col-span-9 space-y-5">
  1147→            {/* Error banner (highest priority in main canvas when present) */}
  1148→            {errorDetails && (
  1149→              <ErrorBanner
  1150→                errorDetails={errorDetails}
  1151→                mode={mode}
  1152→                altLocations={altLocations}
  1153→                onRetry={handleRetry}
  1154→                onSwitchToDemo={() => handleModeChange('FIXTURE')}
  1155→                onSwitchToLive={() => handleModeChange('LIVE')}
  1156→                onSelectAltLocation={handleSelectAltLocation}
  1157→              />
  1158→            )}
  1159→
  1160→            {/* 1. FORTYGUARD THERMAL FIELD — the visual hero */}
  1161→            <ThermalMapCanvas
  1162→              stage={workflowStage}
  1163→              locationName={selectedLocation?.name ?? 'No location selected'}
  1164→              baseTimestamp={spatialFieldMeta?.baseTimestamp}
  1165→              thermalCellCount={thermalCellCount}
  1166→              resolution={selectedLocation ? resolutionDisplay : undefined}
  1167→              mode={mode}
  1168→              loading={loading}
  1169→              selectedLocation={selectedLocation ?? undefined}
  1170→              analysisCenter={selectedLocation ? aoiCenter : undefined}
  1171→              temporalInput={temporalInput}
  1172→              timezone={displayTimezone}
  1173→              rankedCandidates={spatialDecision?.rankedLocations.map((r) => ({
  1174→                locationId: r.locationId,
  1175→                name: r.name,
  1176→                location: r.location,
  1177→              }))}
  1178→              recommendedLocationId={spatialDecision?.recommendedLocation.locationId}
  1179→            >
  1180→              <ThermalMap
  1181→                location={selectedLocation ? { latitude: selectedLocation.latitude, longitude: selectedLocation.longitude } : null}
  1182→                locationState={selectedLocation?.state}
  1183→                locationName={selectedLocation?.name}
  1184→                locationDraggable={mode === 'LIVE'}
  1185→                onMoveOperatingLocation={handleMoveOperatingLocation}
  1186→                regionBoundary={regionBoundary}
  1187→                regionMask={regionMask}
  1188→                regionDisplayName={regionDisplayName}
  1189→                analysisAoi={analysisAoi}
  1190→                aoiInvalid={aoiInvalid}
  1191→                aoiInvalidMessage={aoiValidation.valid ? undefined : aoiValidation.message}
  1192→                captureAoiLabel={mode === 'FIXTURE' && demoCaptureAvailable ? fixtureCaptureSpanLabel() : undefined}
  1193→                spatialField={spatialField}
  1194→                selectedTileId={decision?.evidenceBundle.selectedTileId}
  1195→                candidates={displayCandidates}
  1196→                candidatesDraggable={mode === 'LIVE'}
  1197→                onMoveCandidate={handleMoveCandidate}
  1198→                recommendedLocationId={spatialDecision?.recommendedLocation.locationId}
  1199→                unit={unit}
  1200→                layerVisibility={prefs.mapLayerVisibility}
  1201→                onToggleLayer={prefSetters.setMapLayerVisibility}
  1202→                areaShape={prefs.analysisAreaShape}
  1203→                onMoveAoi={handleMoveAoi}
  1204→                aoiDraggable={mode === 'LIVE'}
  1205→                showLocationMarker={!!selectedLocation}
  1206→                emptyMapMessage={
  1207→                  workflowStage === 'NO_DEMO_CAPTURE'
  1208→                    ? 'No DEMO capture available for this location — switch to LIVE or pick a Manhattan DEMO location'
  1209→                    : 'Select a location to begin the analysis'
  1210→                }
  1211→                addSiteMode={addSiteMode}
  1212→                onAddSiteAt={handleAddSiteAt}
  1213→                cameraBehavior={cameraBehavior}
  1214→                cameraNonce={cameraNonce}
  1215→              />
  1216→            </ThermalMapCanvas>
  1217→
  1218→            {/* 2. RECOMMENDED OPERATION */}
  1219→            {jointDecision && !errorDetails && (
  1220→              <RecommendedOperation
  1221→                jointDecision={jointDecision}
  1222→                unit={unit}
  1223→                timezone={displayTimezone}
  1224→                mode={mode}
  1225→                temporalInput={temporalInput}
  1226→              />
  1227→            )}
  1228→
  1229→            {/* 3. TOP CANDIDATES */}
  1230→            {jointDecision && !errorDetails && (
  1231→              <TopCandidates
  1232→                jointDecision={jointDecision}
  1233→                unit={unit}
  1234→                timezone={displayTimezone}
  1235→              />
  1236→            )}
  1237→
  1238→            {/* 4. WHAT-IF */}
  1239→            {scenarioAnalysis && scenarioAnalysis.scenarios.length > 0 && !errorDetails && (
  1240→              <WhatIfPanel
  1241→                scenarioAnalysis={scenarioAnalysis}
  1242→                selectedScenarioId={selectedScenarioId}
  1243→                onSelectScenario={handleSelectScenario}
  1244→                unit={unit}
  1245→                timezone={displayTimezone}
  1246→              />
  1247→            )}
  1248→
  1249→            {/* 5. GROUNDED AI EXPLANATION (subdued — not the hero) */}
  1250→            {explanation && !errorDetails && (
  1251→              <GroundedExplanation
  1252→                explanation={explanation}
  1253→                unit={unit}
  1254→                timezone={displayTimezone}
  1255→                explaining={explaining}
  1256→                onRefresh={handleRefreshExplanation}
  1257→              />
  1258→            )}
  1259→
  1260→            {/* Empty state before first run (EMPTY / LOCATION_SELECTED stages) */}
  1261→            {!jointDecision && !loading && !errorDetails && (
  1262→              <div
  1263→                className="rounded-xl border border-border bg-surface-card px-5 py-12 text-center"
  1264→                data-testid={workflowStage === 'EMPTY' ? 'empty-workspace-card' : 'pre-generate-card'}
  1265→              >
  1266→                <div className="text-2xl mb-2">🌡</div>
  1267→                {workflowStage === 'EMPTY' ? (
  1268→                  <>
  1269→                    <p className="text-text-primary text-sm font-bold">Select a location to begin</p>
  1270→                    <p className="text-text-muted text-xs mt-1.5 leading-relaxed max-w-md mx-auto">
  1271→                      Search a city, street, or address — or pick a preset in the left rail. Locations with a genuine
  1272→                      captured DEMO dataset load it explicitly; every other location runs LIVE against FortyGuard.
  1273→                    </p>
  1274→                    <p className="text-[10px] font-mono text-text-dimmed mt-3 tracking-wide">
  1275→                      LOCATION → ANALYSIS AOI → THERMAL OBSERVATIONS → CANDIDATES → RECOMMENDATION
  1276→                    </p>
  1277→                  </>
  1278→                ) : (
  1279→                  <>
  1280→                    <p className="text-text-muted text-sm font-medium">Generate a thermal field to see the recommended operational plan.</p>
  1281→                    <p className="text-text-dimmed text-xs mt-1">
  1282→                      {mode === 'LIVE'
  1283→                        ? 'Place candidate sites (+ Site on the map), set the WHEN date/time, then click Generate.'
  1284→                        : 'This location has a genuine captured dataset — Generate replays it (no provider request).'}
  1285→                    </p>
  1286→                  </>
  1287→                )}
  1288→              </div>
  1289→            )}
  1290→          </div>
  1291→        </div>
  1292→      </div>
  1293→
  1294→      {/* SETTINGS DRAWER — provider capability + diagnostics */}
  1295→      <SettingsDrawer open={settingsOpen} onOpenChange={setSettingsOpen} capability={capability} />
  1296→    </main>
  1297→  );
  1298→}
