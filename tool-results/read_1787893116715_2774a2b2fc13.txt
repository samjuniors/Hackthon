   509→            {derivedDuration}h
   510→          </span>
   511→        </div>
   512→
   513→        {/* Inline validation hint */}
   514→        {!allValid && (
   515→          <p className="text-[10.5px] text-destructive mt-1.5 font-medium">
   516→            {!dateValid && 'Enter a valid date (YYYY-MM-DD). '}
   517→            {!rangeValid && 'End time must be after start time.'}
   518→          </p>
   519→        )}
   520→
   521→        {/* Honest future-date note */}
   522→        {mode === 'LIVE' && isFutureDate && (
   523→          <p className="text-[10.5px] mt-1.5 font-medium" style={{ color: 'var(--accent-amber)' }}>
   524→            Future date selected — subject to FortyGuard forecast availability. The provider reports any unsupported
   525→            window verbatim.
   526→          </p>
   527→        )}
   528→
   529→        {/* LIVE billing disclosure */}
   530→        {mode === 'LIVE' && allValid && (
   531→          <div
   532→            className="rounded-lg px-3 py-2 mt-2 border border-emerald-500/25"
   533→            style={{ background: 'var(--accent-emerald-bg)' }}
   534→            data-testid="live-request-disclosure"
   535→          >
   536→            <div className="flex items-center justify-between">
   537→              <span className="text-[10px] font-semibold uppercase tracking-wider" style={{ color: 'var(--accent-emerald)' }}>
   538→                LIVE · {derivedDuration}-hour evaluation
   539→              </span>
   540→              <span className="text-[11px] font-semibold tnum" style={{ color: 'var(--accent-emerald)' }}>
   541→                {liveHourlyRequestCount} FortyGuard hourly request{liveHourlyRequestCount > 1 ? 's' : ''}
   542→              </span>
   543→            </div>
   544→            <p className="text-[10px] text-text-muted mt-1 leading-relaxed">
   545→              Each evaluated hour is its own single-hour FortyGuard request (cached results reused, not re-billed).
   546→              Repeat requests may consume provider credits.
   547→            </p>
   548→          </div>
   549→        )}
   550→
   551→        {/* Human-readable window preview */}
   552→        {allValid && (
   553→          <p className="text-[10.5px] text-text-dimmed tnum mt-2 leading-relaxed">
   554→            {formatTemporalForHeader(temporalInput, tz)}
   555→          </p>
   556→        )}
   557→      </div>
   558→
   559→      <Separator />
   560→
   561→      {/* ── CANDIDATE SITES ── */}
   562→      <div data-testid="candidate-sites-section">
   563→        <SectionLabel
   564→          right={
   565→            <span className="text-[10px] font-medium text-text-dimmed">
   566→              {mode === 'FIXTURE' ? 'DEMO CANDIDATES · application-defined' : 'user-placed'}
   567→            </span>
   568→          }
   569→        >
   570→          Candidate Sites
   571→        </SectionLabel>
   572→
   573→        {mode === 'FIXTURE' ? (
   574→          demoCaptureAvailable ? (
   575→            <div className="space-y-1">
   576→              <p className="text-[10.5px] text-text-muted leading-relaxed">
   577→                Application-defined DEMO candidates — points the application wants evaluated against the captured
   578→                FortyGuard provider data (not captured sites, not FortyGuard observations). LIVE lets you place your
   579→                own sites.
   580→              </p>
   581→              {['Battery Park Greenway', 'City Hall Civic Center', 'Chinatown / Bowery'].map((n) => (
   582→                <div key={n} className="flex items-center gap-2 rounded-md bg-surface-deep px-2.5 py-2">
   583→                  <span className="w-1.5 h-1.5 rounded-full flex-shrink-0" style={{ background: 'var(--text-secondary)' }} />
   584→                  <span className="text-[12px] text-text-secondary">{n}</span>
   585→                  <span className="ml-auto text-[9px] font-mono uppercase text-text-dimmed">demo</span>
   586→                </div>
   587→              ))}
   588→            </div>
   589→          ) : (
   590→            <p className="text-[10.5px] text-text-muted leading-relaxed" data-testid="no-demo-candidates">
   591→              No DEMO candidates — this location has no captured FortyGuard dataset. Switch to LIVE to place your own
   592→              candidate sites, or select a Manhattan DEMO location.
   593→            </p>
   594→          )
   595→        ) : (
   596→          <div className="space-y-2">
   597→            {/* Add-site actions */}
   598→            <div className="grid grid-cols-2 gap-1.5">
   599→              <button
   600→                type="button"
   601→                data-testid="add-site-map-btn"
   602→                onClick={onToggleAddSiteMode}
   603→                aria-pressed={addSiteMode}
   604→                className={`h-9 rounded-lg text-[11.5px] font-medium border flex items-center justify-center gap-1.5 transition-colors duration-150 ${
   605→                  addSiteMode
   606→                    ? 'border-emerald-500 text-emerald-600 dark:text-emerald-400 bg-emerald-500/10'
   607→                    : 'border-border bg-surface-elevated text-text-muted hover:text-text-primary'
   608→                }`}
   609→              >
   610→                <Plus className="size-3.5" aria-hidden="true" />
   611→                {addSiteMode ? 'Click the map…' : 'Add on map'}
   612→              </button>
   613→              <button
   614→                type="button"
   615→                data-testid="add-site-search-btn"
   616→                onClick={() => setShowSiteSearch((v) => !v)}
   617→                aria-pressed={showSiteSearch}
   618→                className={`h-9 rounded-lg text-[11.5px] font-medium border flex items-center justify-center gap-1.5 transition-colors duration-150 ${
   619→                  showSiteSearch
   620→                    ? 'border-primary text-text-primary bg-primary/10'
   621→                    : 'border-border bg-surface-elevated text-text-muted hover:text-text-primary'
   622→                }`}
   623→              >
   624→                <Search className="size-3.5" aria-hidden="true" />
   625→                From search
   626→              </button>
   627→            </div>
   628→
   629→            {/* Site search inline */}
   630→            {showSiteSearch && (
   631→              <div className="rounded-lg border border-border bg-surface-deep p-2">
   632→                <LocationSearch
   633→                  selectedLocation={selectedLocation}
   634→                  mode={mode}
   635→                  onSelectLocation={(loc) => {
   636→                    onAddSiteFromSearch(loc);
   637→                    setShowSiteSearch(false);
   638→                  }}
   639→                  onSwitchToLive={undefined}
   640→                  compact
   641→                  activeStateFilter={activeStateFilter}
   642→                />
   643→                <p className="text-[9.5px] text-text-dimmed mt-1">
   644→                  Only sites inside the analysis area can be added.
   645→                </p>
   646→              </div>
   647→            )}
   648→
   649→            {/* Site list */}
   650→            {candidateSites.length === 0 ? (
   651→              <p className="text-[10.5px] text-text-muted leading-relaxed" data-testid="no-candidate-sites">
   652→                No candidate sites yet. LIVE never fabricates sites — add one on the map or from search, then Generate.
   653→              </p>
   654→            ) : (
   655→              <ul className="space-y-1 max-h-40 overflow-y-auto" data-testid="candidate-sites-list">
   656→                {candidateSites.map((site, idx) => (
   657→                  <li
   658→                    key={site.locationId}
   659→                    className={`rounded-md border px-2.5 py-2 ${
   660→                      site.outsideAoi
   661→                        ? 'border-destructive/50 bg-destructive/5'
   662→                        : 'border-transparent bg-surface-deep'
   663→                    }`}
   664→                  >
   665→                    <div className="flex items-center gap-2">
   666→                      <span
   667→                        className="w-4 h-4 rounded-full flex items-center justify-center text-[9px] font-mono font-bold text-white flex-shrink-0"
   668→                        style={{ backgroundColor: getCandidateColor(idx) }}
   669→                        title={`Site ${idx + 1}`}
   670→                      >
   671→                        {idx + 1}
   672→                      </span>
   673→                      {renamingId === site.locationId ? (
   674→                        <input
   675→                          autoFocus
   676→                          value={renameValue}
   677→                          onChange={(e) => setRenameValue(e.target.value)}
   678→                          onBlur={() => {
   679→                            if (renameValue.trim()) onRenameSite(site.locationId, renameValue.trim());
   680→                            setRenamingId(null);
   681→                          }}
   682→                          onKeyDown={(e) => {
   683→                            if (e.key === 'Enter') {
   684→                              if (renameValue.trim()) onRenameSite(site.locationId, renameValue.trim());
   685→                              setRenamingId(null);
   686→                            }
   687→                          }}
   688→                          className="flex-1 min-w-0 bg-surface-elevated border border-primary/50 rounded px-1.5 py-0.5 text-xs text-text-primary focus:outline-none"
   689→                        />
   690→                      ) : (
   691→                        <button
   692→                          type="button"
   693→                          onClick={() => {
   694→                            setRenamingId(site.locationId);
   695→                            setRenameValue(site.name);
   696→                          }}
   697→                          className="flex-1 min-w-0 text-left text-xs text-text-secondary hover:text-text-primary transition-colors duration-150 truncate"
   698→                          title="Click to rename"
   699→                        >
   700→                          {site.name}
   701→                        </button>
   702→                      )}
   703→                      <span
   704→                        className="text-[9px] font-mono uppercase text-text-dimmed flex-shrink-0"
   705→                        title={`Added via ${site.origin}`}
   706→                      >
   707→                        {site.origin === 'map-click' ? 'map' : site.origin}
   708→                      </span>
   709→                      <button
   710→                        type="button"
   711→                        onClick={() => onRemoveSite(site.locationId)}
   712→                        className="text-text-dimmed hover:text-destructive transition-colors duration-150 text-xs flex-shrink-0"
   713→                        title="Remove site"
   714→                      >
   715→                        ✕
   716→                      </button>
   717→                    </div>
   718→                    {site.outsideAoi && (
   719→                      <div className="text-[9.5px] text-destructive mt-1">
   720→                        Outside the analysis area — move it inside before Generate.
   721→                      </div>
   722→                    )}
   723→                  </li>
   724→                ))}
   725→              </ul>
   726→            )}
   727→            {outsideSiteCount > 0 && (
   728→              <p className="text-[10px] text-destructive" data-testid="outside-sites-warning">
   729→                {outsideSiteCount} site{outsideSiteCount > 1 ? 's' : ''} outside the analysis area — fix before Generate.
   730→              </p>
   731→            )}
   732→          </div>
   733→        )}
   734→      </div>
   735→
   736→      <Separator />
   737→
   738→      {/* ── Selected analysis summary + Generate ── */}
   739→      <div
   740→        className="pt-3 border-t border-border/70 space-y-1"
   741→        data-testid="active-analysis-location-indicator"
   742→      >
   743→        <div className="flex items-center justify-between">
   744→          <span className="text-[10px] font-medium uppercase tracking-wider text-text-dimmed">Selected Analysis Area</span>
   745→          <span
   746→            className="text-[10px] font-semibold"
   747→            style={{ color: mode === 'LIVE' ? 'var(--accent-emerald)' : 'var(--accent-amber)' }}
   748→            data-testid="analysis-mode-badge"
   749→          >
   750→            {mode === 'LIVE' ? 'LIVE · FortyGuard' : 'DEMO · Captured FortyGuard'}
   751→          </span>
   752→        </div>
   753→        <div className="text-[13px] font-semibold text-text-primary leading-tight" data-testid="active-analysis-location-name">
   754→          {selectedLocation ? selectedLocation.name : 'No location selected'}
   755→        </div>
   756→        {selectedLocation && centerCoords && (
   757→          <div className="text-[10.5px] font-mono flex items-center justify-between text-text-muted" data-testid="active-analysis-location-coords">
   758→            <span className="tnum">{centerCoords.latitude.toFixed(4)}°, {centerCoords.longitude.toFixed(4)}°</span>
   759→            <span className="font-sans text-[10px] text-text-dimmed">
   760→              {selectedLocation.city ? `${selectedLocation.city}, ${selectedLocation.state || selectedLocation.country}` : selectedLocation.state || ''}
   761→            </span>
   762→          </div>
   763→        )}
   764→      </div>
   765→
   766→      {/* Generate — the single primary action */}
   767→      <motion.button
   768→        whileTap={{ scale: 0.99 }}
   769→        disabled={loading || generateDisabled}
   770→        onClick={onGenerate}
   771→        data-testid="recalculate-decision-btn"
   772→        title={generateDisabled && !loading ? generateDisabledReason : undefined}
   773→        className={`mt-3 w-full h-12 rounded-xl text-[13px] font-semibold transition-colors duration-150 flex items-center justify-center gap-2 border ${
   774→          loading || generateDisabled
   775→            ? 'bg-surface-deep border-border text-text-dimmed cursor-not-allowed'
   776→            : 'bg-slate-900 hover:bg-slate-800 dark:bg-cyan-400 dark:hover:bg-cyan-300 text-white dark:text-slate-950 border-slate-900 dark:border-cyan-400 cursor-pointer'
   777→        }`}
   778→      >
   779→        {loading ? (
   780→          <span className="flex items-center gap-2">
   781→            <span className="w-4 h-4 border-2 border-white/40 border-t-white rounded-full animate-spin" />
   782→            Generating thermal field…
   783→          </span>
   784→        ) : (
   785→          `Generate Thermal Field · ${mode === 'LIVE' ? 'LIVE' : 'DEMO'}`
   786→        )}
   787→      </motion.button>
   788→
   789→      {/* Inline reason when Generate is blocked */}
   790→      {generateDisabled && !loading && generateDisabledReason && (
   791→        <p
   792→          className="text-[10.5px] leading-relaxed text-center mt-1.5 font-medium"
   793→          style={{ color: 'var(--accent-amber)' }}
   794→          data-testid="generate-blocked-reason"
   795→        >
   796→          {generateDisabledReason}
   797→        </p>
   798→      )}
   799→
   800→      {/* ── SYSTEM STATUS (quiet, at the bottom) ── */}
   801→      <div className="mt-6">
   802→        <SystemStatus
   803→          mode={mode}
   804→          fortyGuardStatus={fortyGuardStatus}
   805→          aiStatus={aiStatus}
   806→          aiProvider={aiProvider}
   807→          fieldReady={fieldReady}
   808→          loading={loading}
   809→          onTestFortyGuard={onTestFortyGuard}
   810→          onTestAI={onTestAI}
   811→        />
   812→      </div>
   813→
   814→      <div className="text-[10px] text-text-dimmed text-center mt-4">
   815→        FortyGuard Hackathon&apos;26
   816→      </div>
   817→    </div>
   818→  );
   819→}
