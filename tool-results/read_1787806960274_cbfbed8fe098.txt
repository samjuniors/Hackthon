   482→            <div className="flex items-center justify-between mb-1.5">
   483→              <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">
   484→                Evaluation Window
   485→              </span>
   486→            </div>
   487→            <div className="grid grid-cols-2 gap-1.5">
   488→              {TIME_MODE_OPTIONS.map((opt) => {
   489→                const active = temporalInput.timeMode === opt.value;
   490→                return (
   491→                  <button
   492→                    key={opt.value}
   493→                    data-testid={`evaluation-window-${opt.value}`}
   494→                    onClick={() => handleTimeModeChange(opt.value)}
   495→                    className={`min-h-[36px] rounded-md text-[10px] font-bold transition-all border leading-tight px-1 ${
   496→                      active
   497→                        ? 'border-accent-cyan bg-accent-cyan-bg text-accent-cyan'
   498→                        : 'border-border bg-surface-elevated text-text-muted hover:text-text-primary hover:bg-surface-deep'
   499→                    }`}
   500→                    aria-pressed={active}
   501→                    title={opt.description}
   502→                  >
   503→                    {opt.label}
   504→                  </button>
   505→                );
   506→              })}
   507→            </div>
   508→            <p className="text-[9px] text-text-dimmed mt-1 leading-relaxed">
   509→              A time range is evaluated as a sequence of hourly FortyGuard requests — one request per hour.
   510→            </p>
   511→          </div>
   512→
   513→          {/* Date input (always visible) */}
   514→          <label className="block mb-2">
   515→            <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">Date</span>
   516→            <input
   517→              type="date"
   518→              value={temporalInput.date}
   519→              onChange={(e) => update({ date: e.target.value })}
   520→              disabled={isFixtureAnchored}
   521→              className={`mt-1 w-full h-10 rounded-lg border bg-surface-elevated px-3 text-sm font-mono text-text-primary focus:outline-none focus:border-accent-cyan transition-colors ${
   522→                dateValid ? 'border-border' : 'border-red-400'
   523→              } ${isFixtureAnchored ? 'opacity-70 cursor-not-allowed' : ''}`}
   524→              aria-label="Analysis date"
   525→            />
   526→            {/* LIVE date hint — honest about model completion, no invented
   527→                "guaranteed valid" date, no silent retry / DEMO fallback. */}
   528→            {mode === 'LIVE' && (
   529→              <span className="block text-[9px] text-text-dimmed mt-1 leading-relaxed" data-testid="live-date-hint">
   530→                Pick the date explicitly. Recently requested periods may have no completed FortyGuard model yet (the
   531→                provider then returns an empty thermal field — reported verbatim, never retried with another date).
   532→                Documented forecast support: up to +12h ahead.
   533→              </span>
   534→            )}
   535→          </label>
   536→
   537→          {/* Start / End time inputs */}
   538→          <div className={`grid ${isSingleHour ? 'grid-cols-1' : 'grid-cols-2'} gap-2 mb-2`}>
   539→            <label className="block">
   540→              <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">
   541→                {isSingleHour ? 'Hour' : 'Start'}
   542→              </span>
   543→              <input
   544→                type="time"
   545→                value={temporalInput.startTime}
   546→                onChange={(e) => update({ startTime: e.target.value })}
   547→                disabled={isFixtureAnchored}
   548→                className={`mt-1 w-full h-10 rounded-lg border bg-surface-elevated px-3 text-sm font-mono text-text-primary focus:outline-none focus:border-accent-cyan transition-colors ${
   549→                  startValid ? 'border-border' : 'border-red-400'
   550→                } ${isFixtureAnchored ? 'opacity-70 cursor-not-allowed' : ''}`}
   551→                aria-label="Analysis start time"
   552→              />
   553→            </label>
   554→            {!isSingleHour && (
   555→              <label className="block">
   556→                <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">End</span>
   557→                <input
   558→                  type="time"
   559→                  value={temporalInput.endTime}
   560→                  onChange={(e) => update({ endTime: e.target.value })}
   561→                  disabled={isFixtureAnchored}
   562→                  className={`mt-1 w-full h-10 rounded-lg border bg-surface-elevated px-3 text-sm font-mono text-text-primary focus:outline-none focus:border-accent-cyan transition-colors ${
   563→                    endValid ? 'border-border' : 'border-red-400'
   564→                  } ${isFixtureAnchored ? 'opacity-70 cursor-not-allowed' : ''}`}
   565→                  aria-label="Analysis end time"
   566→                />
   567→              </label>
   568→            )}
   569→          </div>
   570→
   571→          {/* Derived duration (read-only) */}
   572→          <div className="flex items-center justify-between rounded-lg bg-surface-deep px-3 py-2 border border-border">
   573→            <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">
   574→              Duration
   575→            </span>
   576→            <span
   577→              className="text-base font-black font-mono"
   578→              style={{ color: 'var(--accent-cyan)' }}
   579→              data-testid="duration-display"
   580→            >
   581→              {derivedDuration}h
   582→            </span>
   583→          </div>
   584→
   585→          {/* Inline validation hint */}
   586→          {!allValid && (
   587→            <p className="text-[10px] text-red-400 mt-1.5">
   588→              {!dateValid && 'Enter a valid date (YYYY-MM-DD). '}
   589→              {!rangeValid && 'End time must be after start time.'}
   590→            </p>
   591→          )}
   592→
   593→          {/* Honest future-date note */}
   594→          {mode === 'LIVE' && isFutureDate && (
   595→            <p className="text-[10px] mt-1.5" style={{ color: 'var(--accent-amber)' }}>
   596→              Future date selected — subject to FortyGuard forecast availability. The provider reports any unsupported window verbatim.
   597→            </p>
   598→          )}
   599→
   600→          {/* LIVE billing disclosure — the engine sends ONE hourly provider
   601→              request per evaluated hour; make that explicit BEFORE Generate.
   602→              Conservative wording: no exact credit cost is claimed. */}
   603→          {mode === 'LIVE' && allValid && (
   604→            <div
   605→              className="rounded-lg px-3 py-2 mt-2 border"
   606→              style={{ background: 'rgba(5,150,105,0.08)', borderColor: 'rgba(5,150,105,0.35)' }}
   607→              data-testid="live-request-disclosure"
   608→            >
   609→              <div className="flex items-center justify-between">
   610→                <span className="text-[9px] font-bold uppercase tracking-widest" style={{ color: '#059669' }}>
   611→                  LIVE · {derivedDuration}-hour evaluation
   612→                </span>
   613→                <span className="text-[10px] font-mono font-bold" style={{ color: '#059669' }}>
   614→                  {liveHourlyRequestCount} FortyGuard hourly request{liveHourlyRequestCount > 1 ? 's' : ''}
   615→                </span>
   616→              </div>
   617→              <p className="text-[9px] text-text-muted mt-0.5 leading-relaxed">
   618→                Each evaluated hour is submitted as its own single-hour FortyGuard /v1/heatmap request (cached results are
   619→                reused, not re-billed). Repeat requests may consume provider credits.
   620→              </p>
   621→            </div>
   622→          )}
   623→
   624→          {/* Human-readable window preview */}
   625→          {allValid && (
   626→            <p className="text-[10px] font-mono text-text-muted mt-2 leading-relaxed">
   627→              {formatTemporalForHeader(temporalInput, tz)}
   628→            </p>
   629→          )}
   630→        </div>
   631→
   632→        {/* Selected Analysis Area indicator — coordinates track AOI drag */}
   633→        <div
   634→          className="rounded-lg p-3 bg-surface-deep border border-border space-y-1"
   635→          data-testid="active-analysis-location-indicator"
   636→        >
   637→          <div className="flex items-center justify-between">
   638→            <span className="text-[9px] font-bold uppercase tracking-widest text-text-dimmed">Selected Analysis Area</span>
   639→            <span
   640→              className="px-2 py-0.5 rounded text-[10px] font-bold border"
   641→              style={mode === 'LIVE'
   642→                ? { background: 'var(--accent-cyan-bg)', color: 'var(--accent-cyan)', borderColor: 'var(--accent-cyan)' }
   643→                : { background: 'var(--accent-amber-bg)', color: 'var(--accent-amber)', borderColor: 'var(--accent-amber)' }
   644→              }
   645→              data-testid="analysis-mode-badge"
   646→            >
   647→              {mode === 'LIVE' ? 'LIVE · FortyGuard' : 'DEMO · Captured FortyGuard'}
   648→            </span>
   649→          </div>
   650→          <div className="text-sm font-bold text-text-primary leading-tight" data-testid="active-analysis-location-name">
   651→            {selectedLocation.name}
   652→          </div>
   653→          <div className="text-[11px] font-mono flex items-center justify-between" style={{ color: 'var(--accent-cyan)' }} data-testid="active-analysis-location-coords">
   654→            <span>{centerCoords.latitude.toFixed(4)}°, {centerCoords.longitude.toFixed(4)}°</span>
   655→            <span className="text-text-muted font-sans text-[10px]">
   656→              {selectedLocation.city ? `${selectedLocation.city}, ${selectedLocation.state || selectedLocation.country}` : selectedLocation.state || ''}
   657→            </span>
   658→          </div>
   659→        </div>
   660→
   661→        {/* Generate button */}
   662→        <motion.button
   663→          whileTap={{ scale: 0.99 }}
   664→          disabled={loading}
   665→          onClick={onGenerate}
   666→          data-testid="recalculate-decision-btn"
   667→          className={`w-full h-12 rounded-xl text-sm font-bold transition-all flex items-center justify-center gap-2 ${
   668→            loading ? 'bg-surface-elevated text-text-dimmed cursor-not-allowed' : 'text-white hover:scale-[1.01] active:scale-[0.99]'
   669→          }`}
   670→          style={loading ? {} : {
   671→            background: mode === 'LIVE'
   672→              ? 'linear-gradient(135deg, var(--accent-emerald), #0d9488)'
   673→              : 'linear-gradient(135deg, var(--accent-cyan), #0284c7)',
   674→            boxShadow: mode === 'LIVE'
   675→              ? '0 4px 16px rgba(5,150,105,0.3)'
   676→              : '0 4px 16px rgba(14,165,233,0.3)',
   677→          }}
   678→        >
   679→          {loading ? (
   680→            <span className="flex items-center gap-2">
   681→              <span className="w-4 h-4 border-2 border-text-dimmed rounded-full animate-spin" style={{ borderTopColor: 'var(--accent-cyan)' }} />
   682→              Generating thermal field…
   683→            </span>
   684→          ) : (
   685→            `⚡ Generate Thermal Field (${mode === 'LIVE' ? 'LIVE' : 'DEMO'})`
   686→          )}
   687→        </motion.button>
   688→      </div>
   689→
   690→      {/* ── SYSTEM STATUS ── */}
   691→      <SystemStatus
   692→        mode={mode}
   693→        fortyGuardStatus={fortyGuardStatus}
   694→        aiStatus={aiStatus}
   695→        aiProvider={aiProvider}
   696→        fieldReady={fieldReady}
   697→        loading={loading}
   698→        onTestFortyGuard={onTestFortyGuard}
   699→        onTestAI={onTestAI}
   700→      />
   701→
   702→      <div className="text-[10px] text-text-dimmed font-mono text-center">
   703→        FortyGuard Hackathon&apos;26
   704→      </div>
   705→    </div>
   706→  );
   707→}
