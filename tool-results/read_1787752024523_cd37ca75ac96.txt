     3→ * FortyGuard raw fixture capture — evidence gate tooling (M4 / Phase 1).
     4→ *
     5→ * Zero runtime dependencies (Node >= 18 native fetch + fs).
     6→ *
     7→ * PURPOSE
     8→ *   Capture *raw, unmodified* FortyGuard API responses to tests/fixtures/ so the
     9→ *   decision engine can be built against observed schemas rather than assumed
    10→ *   ones. Every capture writes two files:
    11→ *
    12→ *     <name>.json          the raw response body, verbatim, no wrapper
    13→ *     <name>.request.json  the request we sent + timing/status metadata
    14→ *
    15→ * CREDENTIALS
    16→ *   Read from the environment only. Never hardcoded, never written to a fixture,
    17→ *   never printed. The API key is redacted from all output including error text.
    18→ *
    19→ *     FORTYGUARD_API_KEY        (required)
    20→ *     FORTYGUARD_API_BASE_URL   (optional, defaults to https://api.fortyguard.com)
    21→ *
    22→ *   The script loads .env.local then .env if the vars are not already set.
    23→ *
    24→ * USAGE
    25→ *   node scripts/capture-fixtures.mjs usage
    26→ *   node scripts/capture-fixtures.mjs heatmap-ft1
    27→ *   node scripts/capture-fixtures.mjs heatmap-ft2
    28→ *   node scripts/capture-fixtures.mjs env-params
    29→ *   node scripts/capture-fixtures.mjs gate3-sweep
    30→ *   node scripts/capture-fixtures.mjs all

    32→ * CREDIT COST
    33→ *   /v1/heatmap and /v1/env_params cost 2,000 credits per call.

    47→function loadEnvFile(name) {
    48→  const path = resolve(ROOT, name);
    49→  if (!existsSync(path)) return;
    50→  for (const line of readFileSync(path, "utf8").split(/\r?\n/)) {
    51→    const m = /^\s*([A-Za-z_][A-Za-z0-9_]*)\s*=\s*(.*)\s*$/.exec(line);
    52→    if (!m) continue;
    53→    const [, key, rawValue] = m;
    54→    if (process.env[key]) continue;
    55→    const value = rawValue.replace(/^(['"])(.*)\1$/, "$2");
    56→    if (value) process.env[key] = value;
    57→  }
    58→}

    84→/** Next whole UTC hour, offset by `plusHours`. */
    85→function utcHour(plusHours = 0) {
    86→  const d = new Date();
    87→  d.setUTCMinutes(0, 0, 0);
    88→  d.setUTCHours(d.getUTCHours() + plusHours);
    89→  return {
    90→    date: `${d.getUTCFullYear()}-${pad(d.getUTCMonth() + 1)}-${pad(d.getUTCDate())}`,
    91→    time: `${pad(d.getUTCHours())}:00`,
    92→    iso: d.toISOString(),
    93→  };
    94→}

    96→/** A specific UTC hour on a date `daysBack` before today. */
    97→function pastDayHour(daysBack, utcHourOfDay) {
    98→  const d = new Date();
    99→  d.setUTCDate(d.getUTCDate() - daysBack);
   100→  d.setUTCMinutes(0, 0, 0);
   101→  d.setUTCHours(utcHourOfDay);
   102→  return {
   103→    start_date: `${d.getUTCFullYear()}-${pad(d.getUTCMonth() + 1)}-${pad(d.getUTCDate())}`,
   104→    start_time: `${pad(utcHourOfDay)}:00`,
   105→    filter_type: 1,
   106→  };
   107→}

   109→/** Noon UTC on a date `daysBack` before today. */
   110→function pastDayNoon(daysBack) {
   111→  return pastDayHour(daysBack, 12);

   114→/**
   115→ * Rectangular AOI polygon around a centre point, with independent half-extents.
   116→ * The candidate sites span ~1.7 km of longitude but share one latitude, so a
   117→ * rectangle covers them with far fewer tiles than the enclosing square.
   118→ */
   119→function rectAoi(latitude, longitude, halfLonMetres, halfLatMetres) {
   120→  const dLat = halfLatMetres / 111320;
   121→  const dLon = halfLonMetres / (111320 * Math.cos((latitude * Math.PI) / 180));
   122→  const ring = [
   123→    [longitude - dLon, latitude - dLat],
   124→    [longitude + dLon, latitude - dLat],
   125→    [longitude + dLon, latitude + dLat],
   126→    [longitude - dLon, latitude + dLat],
   127→    [longitude - dLon, latitude - dLat],
   128→  ];
   129→  return {
   130→    type: "FeatureCollection",
   131→    features: [{ type: "Feature", properties: {}, geometry: { type: "Polygon", coordinates: [ring] } }],
   132→  };
   133→}

   135→/**
   136→ * Square-ish AOI polygon around a centre point.
   137→ * GeoJSON ring order is [longitude, latitude] and the ring must be closed.
   138→ */
   139→function squareAoi(latitude, longitude, halfSideMetres) {
   140→  const dLat = halfSideMetres / 111_320;
   141→  const dLon = halfSideMetres / (111_320 * Math.cos((latitude * Math.PI) / 180));
   142→  const ring = [
   143→    [longitude - dLon, latitude - dLat],
   144→    [longitude + dLon, latitude - dLat],
   145→    [longitude + dLon, latitude + dLat],
   146→    [longitude - dLon, latitude + dLat],
   147→    [longitude - dLon, latitude - dLat],
   148→  ];
   149→  return {
   150→    type: "FeatureCollection",
   151→    features: [{ type: "Feature", properties: {}, geometry: { type: "Polygon", coordinates: [ring] } }],
   152→  };
   153→}

   155→function save(name, raw, meta) {
   156→  mkdirSync(FIXTURES, { recursive: true });
   157→  // Raw response body, verbatim — no wrapper, so the observed shape is preserved.
   158→  writeFileSync(resolve(FIXTURES, `${name}.json`), JSON.stringify(raw, null, 2) + "\n", "utf8");
   159→  writeFileSync(resolve(FIXTURES, `${name}.request.json`), JSON.stringify(meta, null, 2) + "\n", "utf8");
   160→  log(`  saved tests/fixtures/${name}.json  (+ .request.json)`);
   161→}

   163→async function call(method, path, body) {
   164→  const url = `${BASE_URL}${path}`;
   165→  const startedAt = Date.now();
   166→  const res = await fetch(url, {
   167→    method,
   168→    headers: {
   169→      "api-key": API_KEY,
   170→      accept: "application/json",
   171→      ...(body ? { "content-type": "application/json" } : {}),
   172→    },
   173→    ...(body ? { body: JSON.stringify(body) } : {}),
   174→  });
   175→  const text = await res.text();
   176→  let parsed;
   177→  try {
   178→    parsed = JSON.parse(text);
   179→  } catch {
   180→    parsed = { __nonJsonBody: text.slice(0, 4000) };
   181→  }
   182→  return { status: res.status, ok: res.ok, body: parsed, elapsedMs: Date.now() - startedAt };
   183→}

   185→/** Submit → poll /v1/status/{activity_id} until terminal. Returns both stages. */
   186→async function submitAndPoll(path, requestBody, { maxAttempts = 120, intervalMs = 3000 } = {}) {
   187→  log(`  POST ${path}`);
   188→  const submit = await call("POST", path, requestBody);
   189→  if (!submit.ok) {
   190→    log(`  submit failed: HTTP ${submit.status} ${JSON.stringify(submit.body).slice(0, 400)}`);
   191→    return { submit, polls: [], final: null };
   192→  }
   193→
   194→  const activityId =
   195→    submit.body?.data?.activity_id ?? submit.body?.activity_id ?? submit.body?.data?.activityId ?? null;
   196→  log(`  activity_id: ${activityId ?? "(none returned)"}`);
   197→  if (!activityId) return { submit, polls: [], final: null };
   198→
   199→  const polls = [];
   200→  for (let attempt = 1; attempt <= maxAttempts; attempt++) {
   201→    await new Promise((r) => setTimeout(r, intervalMs));
   202→    const poll = await call("GET", `/v1/status/${activityId}`);
   203→    const status = poll.body?.data?.status ?? poll.body?.status ?? "(unknown)";
   204→    polls.push({ attempt, httpStatus: poll.status, status, elapsedMs: poll.elapsedMs });
   205→    log(`  poll ${attempt}: ${status}`);
   206→    if (status === "Completed" || status === "Failed") {
   207→      return { submit, polls, final: poll, activityId };
   208→    }
   209→  }
   210→  log(`  gave up after ${maxAttempts} polls`);
   211→  return { submit, polls, final: null, activityId };
   212→}

   320→/**
   321→ * GATE 3 — is `temperature` a true reference anchor?
   322→ *
   323→ * Hold latitude/longitude/timestamp fixed and vary ONLY `temperature`. If the
   324→ * returned wet-bulb / heat-index / apparent-temperature values move with the
   325→ * input, the parameter genuinely drives the physics. If they do not move, the
   326→ * input is ignored or merely advisory and a per-tile "wet-bulb field" built by
   327→ * feeding tile temperatures into this endpoint would be fabricated.
   328→ */
   329→async function captureGate3Sweep(temperatures = [24, 38]) {
   330→  log("\n[gate3-sweep] GATE 3 — temperature sensitivity — 2,000 credits per probe");
   331→  const hour = utcHour(2);
   332→  const probes = [];
   333→  for (const temperature of temperatures) {
   334→    log(`  probe temperature=${temperature}`);
   335→    const request = {
   336→      latitude: CENTRE.latitude,
   337→      longitude: CENTRE.longitude,
   338→      temperature,
   339→      date_time: { start_date: hour.date, start_time: hour.time, filter_type: 1 },
   340→      analysis: [
   341→        "heat_index_celsius",
   342→        "apparent_temperature_celsius",
   343→        "wet_bulb_temperature_celsius",
   344→        "relative_humidity_percent",
   345→      ],
   346→    };
   347→    const result = await submitAndPoll("/v1/env_params", request);
   348→    const name = `env_params_gate3_t${temperature}`;
   349→    save(name, result.final?.body ?? { __submitOnly: result.submit.body }, {
   350→      endpoint: "/v1/env_params",
   351→      gate: "GATE 3 — vary temperature only, hold location+time fixed",
   352→      requestBody: request,
   353→      intendedUtcHour: hour.iso,
   354→      activityId: result.activityId ?? null,
   355→      submitHttpStatus: result.submit.status,
   356→      polls: result.polls,
   357→      capturedAt: new Date().toISOString(),
   358→    });
   359→    probes.push({ temperature, fixture: name, body: result.final?.body ?? null });
   360→  }
   361→  return probes;
   362→}

   364→/**
   365→ * GATE 4 — why does a Completed /v1/heatmap return `n_cells: 0`?
   366→ *
   367→ * The baseline capture (heatmap_ft1_raw.json) reached status `Completed` and
   368→ * still returned `map_data.features: []` with `stats_data.n_cells: 0`. A
   369→ * Completed-but-empty result narrows the cause to how the request describes the
   370→ * area, not to auth, timing, or coverage. Each variant below changes exactly one
   371→ * thing so the responsible parameter is identifiable from the response alone.
   372→ *
   373→ * Each probe is a billable call (~4,200 credits observed). Variants are ordered
   374→ * cheapest-hypothesis-first; stop as soon as one returns cells.

   376→function bareGeometry(latitude, longitude, halfSideMetres) {

   545→/** Count tiles in whatever shape the response returns them. */
   546→function summarizeHeatmapResult(body) {
   547→  const result = body?.data?.result ?? {};
   548→  const mapData = result.map_data ?? result;
   549→  const features = Array.isArray(mapData?.features) ? mapData.features : [];
   550→  const stats = result.stats_data ?? {};
   551→  const sampleProps = features[0]?.properties ?? null;
   552→  return {
   553→    featureCount: features.length,
   554→    nCells: stats.n_cells ?? null,
   555→    statsKeys: Object.keys(stats),
   556→    samplePropertyKeys: sampleProps ? Object.keys(sampleProps) : [],
   557→    sampleProperties: sampleProps,
   558→    sampleGeometryType: features[0]?.geometry?.type ?? null,
   559→    sampleRing: features[0]?.geometry?.coordinates?.[0] ?? null,
   560→  };
   561→}

   563→async function captureHeatmapProbe() {
   564→  const requested = process.argv.slice(3);
   565→  const names = requested.length > 0 ? requested : Object.keys(HEATMAP_PROBES);
   566→
   567→  log("\n[heatmap-probe] GATE 4 — diagnosing Completed-but-empty heatmap responses");
   568→  const summaries = [];
   569→
   570→  for (const name of names) {
   571→    const probe = HEATMAP_PROBES[name];
   572→    if (!probe) {
   573→      console.error(`unknown probe "${name}". known: ${Object.keys(HEATMAP_PROBES).join(", ")}`);
   574→      process.exit(1);
   575→    }
   576→
   577→    log(`\n  --- probe ${name} — ${probe.hypothesis}`);
   578→    const hour = utcHour(2);
   579→    const request = {
   580→      polygon_aoi: probe.aoi(),
   581→      date_time: probe.dateTime
   582→        ? probe.dateTime()
   583→        : { start_date: hour.date, start_time: hour.time, filter_type: 1 },
   584→      granularity: probe.granularity,
   585→      ...(probe.analyticType ? { analytic_type: probe.analyticType } : {}),
   586→    };
   587→
   588→    const result = await submitAndPoll("/v1/heatmap", request);
   589→    const body = result.final?.body ?? { __submitOnly: result.submit.body };
   590→    const summary = summarizeHeatmapResult(body);
   591→
   592→    save(`heatmap_probe_${name.replace(/-/g, "_")}`, body, {
   593→      endpoint: "/v1/heatmap",
   594→      gate: `GATE 4 — ${probe.hypothesis}`,
   595→      probe: name,
   596→      requestBody: request,
   597→      intendedUtcHour: hour.iso,
   598→      activityId: result.activityId ?? null,
   599→      submitHttpStatus: result.submit.status,
   600→      polls: result.polls,
   601→      responseSummary: summary,
   602→      capturedAt: new Date().toISOString(),
   603→    });
   604→
   605→    log(`      features: ${summary.featureCount}  n_cells: ${summary.nCells}`);
   606→    if (summary.samplePropertyKeys.length > 0) {
   607→      log(`      tile properties: ${summary.samplePropertyKeys.join(", ")}`);
   608→      log(`      first tile: ${JSON.stringify(summary.sampleProperties)}`);
   609→    }
   610→
   611→    summaries.push({ name, ...summary });
   612→
   613→    if (summary.featureCount > 0 && probe.stopOnSuccess !== false) {
   614→      log(`\n  ✅ probe "${name}" returned ${summary.featureCount} tiles — hypothesis confirmed, stopping sweep.`);
   615→      break;
   616→    }
   617→  }
   618→
   619→  log("\n  probe summary:");
   620→  for (const s of summaries) {
   621→    log(`    ${s.name.padEnd(18)} features=${String(s.featureCount).padEnd(5)} n_cells=${s.nCells}`);
   622→  }
   623→  return summaries;
   624→}
