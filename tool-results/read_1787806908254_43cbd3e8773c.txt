    81→    81→
    82→    82→async function safeJsonFetch<T = Record<string, unknown>>(
    83→    83→  url: string,
    84→    84→  init?: RequestInit
    85→    85→): Promise<{ ok: boolean; data: T | null }> {
    86→    86→  try {
    87→    87→    const res = await fetch(url, init);
    88→    88→    const text = await res.text();
    89→    89→    if (!text || !text.trim()) return { ok: res.ok, data: null };
    90→    90→    try {
    91→    91→      return { ok: res.ok, data: JSON.parse(text) as T };
    92→    92→    } catch {
    93→    93→      return { ok: false, data: null };
    94→    94→    }
    95→    95→  } catch {
    96→    96→    return { ok: false, data: null };
    97→    97→  }
    98→    98→}
