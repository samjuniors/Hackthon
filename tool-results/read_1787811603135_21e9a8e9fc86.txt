   477→                          className="text-[8px] font-mono uppercase text-text-dimmed flex-shrink-0"
   478→                          title={`Added via ${site.origin}`}
   479→                        >
   480→                          {site.origin === 'map-click' ? 'map' : site.origin}
   481→                        </span>
   482→                        <button
   483→                          type="button"
   484→                          onClick={() => onRemoveSite(site.locationId)}
   485→                          className="text-text-dimmed hover:text-red-400 transition-colors text-xs flex-shrink-0"
   486→                          title="Remove site"
   487→                        >
   488→                          ✕
   489→                        </button>
   490→                      </div>
   491→                      {site.outsideAoi && (
   492→                        <div className="text-[9px] text-red-400 mt-0.5">
   493→                          Outside the analysis area — move it inside or drag the AOI to cover it.
   494→                        </div>
   495→                      )}
   496→                    </li>
   497→                  ))}
   498→                </ul>
   499→              )}
   500→              {outsideSiteCount > 0 && (
   501→                <p className="text-[9px] text-red-400" data-testid="outside-sites-warning">
   502→                  {outsideSiteCount} site{outsideSiteCount > 1 ? 's' : ''} outside the analysis area — fix before Generate.
   503→                </p>
   504→              )}
   505→            </div>
   506→          )}
   507→        </div>
   508→
   509→        <div className="border-t border-border" />
   510→
   511→        {/* ── WHEN — explicit date + time window (Section 14) ── */}
   512→        <div data-testid="when-section">
   513→          <div className="flex items-center justify-between mb-2">
   514→            <span className="text-sm font-medium text-text-secondary">WHEN</span>
   515→            {isFixtureAnchored ? (
   516→              <span
   517→                className="px-2 py-0.5 rounded text-[9px] font-mono font-bold border"
   518→                style={{
   519→                  background: 'var(--accent-amber-bg)',
   520→                  color: 'var(--accent-amber)',
   521→                  borderColor: 'var(--accent-amber)',
   522→                }}
   523→                title={FIXTURE_TEMPORAL_METADATA.captureLabel}
   524→              >
   525→                Fixture capture
   526→              </span>
   527→            ) : (
   528→              <span className="text-[10px] font-mono text-text-dimmed">{tz || 'UTC'}</span>
   529→            )}
   530→          </div>
   531→
   532→          {/* EVALUATION WINDOW selector — a UI concept, NOT a provider
   533→              filter_type. The verified wire contract: every evaluated hour is
   534→              its own single-hour FortyGuard request (filter_type: 1). */}
   535→          <div className="mb-2.5">
   536→            <div className="flex items-center justify-between mb-1.5">
   537→              <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">
   538→                Evaluation Window
   539→              </span>
   540→            </div>
   541→            <div className="grid grid-cols-2 gap-1.5">
   542→              {TIME_MODE_OPTIONS.map((opt) => {
   543→                const active = temporalInput.timeMode === opt.value;
   544→                return (
   545→                  <button
   546→                    key={opt.value}
   547→                    data-testid={`evaluation-window-${opt.value}`}
   548→                    onClick={() => handleTimeModeChange(opt.value)}
   549→                    className={`min-h-[36px] rounded-md text-[10px] font-bold transition-all border leading-tight px-1 ${
   550→                      active
   551→                        ? 'border-accent-cyan bg-accent-cyan-bg text-accent-cyan'
   552→                        : 'border-border bg-surface-elevated text-text-muted hover:text-text-primary hover:bg-surface-deep'
   553→                    }`}
   554→                    aria-pressed={active}
   555→                    title={opt.description}
   556→                  >
   557→                    {opt.label}
   558→                  </button>
   559→                );
   560→              })}
   561→            </div>
   562→            <p className="text-[9px] text-text-dimmed mt-1 leading-relaxed">
   563→              A time range is evaluated as a sequence of hourly FortyGuard requests — one request per hour.
   564→            </p>
   565→          </div>
   566→
   567→          {/* Date input (always visible) */}
   568→          <label className="block mb-2">
   569→            <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">Date</span>
   570→            <input
   571→              type="date"
   572→              value={temporalInput.date}
   573→              onChange={(e) => update({ date: e.target.value })}
   574→              disabled={isFixtureAnchored}
   575→              className={`mt-1 w-full h-10 rounded-lg border bg-surface-elevated px-3 text-sm font-mono text-text-primary focus:outline-none focus:border-accent-cyan transition-colors ${
   576→                dateValid ? 'border-border' : 'border-red-400'
   577→              } ${isFixtureAnchored ? 'opacity-70 cursor-not-allowed' : ''}`}
   578→              aria-label="Analysis date"
   579→            />
   580→            {/* LIVE date hint — honest about model completion, no invented
   581→                "guaranteed valid" date, no silent retry / DEMO fallback. */}
   582→            {mode === 'LIVE' && (
   583→              <span className="block text-[9px] text-text-dimmed mt-1 leading-relaxed" data-testid="live-date-hint">
   584→                Pick the date explicitly. Recently requested periods may have no completed FortyGuard model yet (the
   585→                provider then returns an empty thermal field — reported verbatim, never retried with another date).
   586→                Documented forecast support: up to +12h ahead.
   587→              </span>
   588→            )}
   589→          </label>
   590→
   591→          {/* Start / End time inputs */}
   592→          <div className={`grid ${isSingleHour ? 'grid-cols-1' : 'grid-cols-2'} gap-2 mb-2`}>
   593→            <label className="block">
   594→              <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">
   595→                {isSingleHour ? 'Hour' : 'Start'}
   596→              </span>
   597→              <input
   598→                type="time"
   599→                value={temporalInput.startTime}
   600→                onChange={(e) => update({ startTime: e.target.value })}
   601→                disabled={isFixtureAnchored}
   602→                className={`mt-1 w-full h-10 rounded-lg border bg-surface-elevated px-3 text-sm font-mono text-text-primary focus:outline-none focus:border-accent-cyan transition-colors ${
   603→                  startValid ? 'border-border' : 'border-red-400'
   604→                } ${isFixtureAnchored ? 'opacity-70 cursor-not-allowed' : ''}`}
   605→                aria-label="Analysis start time"
   606→              />
   607→            </label>
   608→            {!isSingleHour && (
   609→              <label className="block">
   610→                <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">End</span>
   611→                <input
   612→                  type="time"
   613→                  value={temporalInput.endTime}
   614→                  onChange={(e) => update({ endTime: e.target.value })}
   615→                  disabled={isFixtureAnchored}
   616→                  className={`mt-1 w-full h-10 rounded-lg border bg-surface-elevated px-3 text-sm font-mono text-text-primary focus:outline-none focus:border-accent-cyan transition-colors ${
   617→                    endValid ? 'border-border' : 'border-red-400'
   618→                  } ${isFixtureAnchored ? 'opacity-70 cursor-not-allowed' : ''}`}
   619→                  aria-label="Analysis end time"
   620→                />
   621→              </label>
   622→            )}
   623→          </div>
   624→
   625→          {/* Derived duration (read-only) */}
   626→          <div className="flex items-center justify-between rounded-lg bg-surface-deep px-3 py-2 border border-border">
   627→            <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">
   628→              Duration
   629→            </span>
   630→            <span
   631→              className="text-base font-black font-mono"
   632→              style={{ color: 'var(--accent-cyan)' }}
   633→              data-testid="duration-display"
   634→            >
   635→              {derivedDuration}h
   636→            </span>
   637→          </div>
   638→
   639→          {/* Inline validation hint */}
   640→          {!allValid && (
   641→            <p className="text-[10px] text-red-400 mt-1.5">
   642→              {!dateValid && 'Enter a valid date (YYYY-MM-DD). '}
   643→              {!rangeValid && 'End time must be after start time.'}
   644→            </p>
   645→          )}
   646→
   647→          {/* Honest future-date note */}
   648→          {mode === 'LIVE' && isFutureDate && (
   649→            <p className="text-[10px] mt-1.5" style={{ color: 'var(--accent-amber)' }}>
   650→              Future date selected — subject to FortyGuard forecast availability. The provider reports any unsupported window verbatim.
   651→            </p>
   652→          )}
   653→
   654→          {/* LIVE billing disclosure — the engine sends ONE hourly provider
   655→              request per evaluated hour; make that explicit BEFORE Generate.
   656→              Conservative wording: no exact credit cost is claimed. */}
   657→          {mode === 'LIVE' && allValid && (
   658→            <div
   659→              className="rounded-lg px-3 py-2 mt-2 border"
   660→              style={{ background: 'rgba(5,150,105,0.08)', borderColor: 'rgba(5,150,105,0.35)' }}
   661→              data-testid="live-request-disclosure"
   662→            >
   663→              <div className="flex items-center justify-between">
   664→                <span className="text-[9px] font-bold uppercase tracking-widest" style={{ color: '#059669' }}>
   665→                  LIVE · {derivedDuration}-hour evaluation
   666→                </span>
   667→                <span className="text-[10px] font-mono font-bold" style={{ color: '#059669' }}>
   668→                  {liveHourlyRequestCount} FortyGuard hourly request{liveHourlyRequestCount > 1 ? 's' : ''}
   669→                </span>
   670→              </div>
   671→              <p className="text-[9px] text-text-muted mt-0.5 leading-relaxed">
   672→                Each evaluated hour is submitted as its own single-hour FortyGuard /v1/heatmap request (cached results are
   673→                reused, not re-billed). Repeat requests may consume provider credits.
   674→              </p>
   675→            </div>
   676→          )}
   677→
   678→          {/* Human-readable window preview */}
   679→          {allValid && (
   680→            <p className="text-[10px] font-mono text-text-muted mt-2 leading-relaxed">
   681→              {formatTemporalForHeader(temporalInput, tz)}
   682→            </p>
   683→          )}
   684→        </div>
   685→
   686→        {/* Selected Analysis Area indicator — coordinates track AOI drag */}
   687→        <div
   688→          className="rounded-lg p-3 bg-surface-deep border border-border space-y-1"
   689→          data-testid="active-analysis-location-indicator"
   690→        >
   691→          <div className="flex items-center justify-between">
   692→            <span className="text-[9px] font-bold uppercase tracking-widest text-text-dimmed">Selected Analysis Area</span>
   693→            <span
   694→              className="px-2 py-0.5 rounded text-[10px] font-bold border"
   695→              style={mode === 'LIVE'
   696→                ? { background: 'var(--accent-cyan-bg)', color: 'var(--accent-cyan)', borderColor: 'var(--accent-cyan)' }
   697→                : { background: 'var(--accent-amber-bg)', color: 'var(--accent-amber)', borderColor: 'var(--accent-amber)' }
   698→              }
   699→              data-testid="analysis-mode-badge"
   700→            >
   701→              {mode === 'LIVE' ? 'LIVE · FortyGuard' : 'DEMO · Captured FortyGuard'}
   702→            </span>
   703→          </div>
   704→          <div className="text-sm font-bold text-text-primary leading-tight" data-testid="active-analysis-location-name">
   705→            {selectedLocation ? selectedLocation.name : 'No location selected'}
   706→          </div>
   707→          {selectedLocation && centerCoords && (
   708→            <div className="text-[11px] font-mono flex items-center justify-between" style={{ color: 'var(--accent-cyan)' }} data-testid="active-analysis-location-coords">
   709→              <span>{centerCoords.latitude.toFixed(4)}°, {centerCoords.longitude.toFixed(4)}°</span>
   710→              <span className="text-text-muted font-sans text-[10px]">
   711→                {selectedLocation.city ? `${selectedLocation.city}, ${selectedLocation.state || selectedLocation.country}` : selectedLocation.state || ''}
   712→              </span>
   713→            </div>
   714→          )}
   715→        </div>
   716→
   717→        {/* Generate button */}
   718→        <motion.button
   719→          whileTap={{ scale: 0.99 }}
   720→          disabled={loading}
   721→          onClick={onGenerate}
   722→          data-testid="recalculate-decision-btn"
   723→          className={`w-full h-12 rounded-xl text-sm font-bold transition-all flex items-center justify-center gap-2 ${
   724→            loading ? 'bg-surface-elevated text-text-dimmed cursor-not-allowed' : 'text-white hover:scale-[1.01] active:scale-[0.99]'
   725→          }`}
   726→          style={loading ? {} : {
   727→            background: mode === 'LIVE'
   728→              ? 'linear-gradient(135deg, var(--accent-emerald), #0d9488)'
   729→              : 'linear-gradient(135deg, var(--accent-cyan), #0284c7)',
   730→            boxShadow: mode === 'LIVE'
   731→              ? '0 4px 16px rgba(5,150,105,0.3)'
   732→              : '0 4px 16px rgba(14,165,233,0.3)',
   733→          }}
   734→        >
   735→          {loading ? (
   736→            <span className="flex items-center gap-2">
   737→              <span className="w-4 h-4 border-2 border-text-dimmed rounded-full animate-spin" style={{ borderTopColor: 'var(--accent-cyan)' }} />
   738→              Generating thermal field…
   739→            </span>
   740→          ) : (
   741→            `⚡ Generate Thermal Field (${mode === 'LIVE' ? 'LIVE' : 'DEMO'})`
   742→          )}
   743→        </motion.button>
   744→      </div>
   745→
   746→      {/* ── SYSTEM STATUS ── */}
   747→      <SystemStatus
   748→        mode={mode}
   749→        fortyGuardStatus={fortyGuardStatus}
   750→        aiStatus={aiStatus}
   751→        aiProvider={aiProvider}
   752→        fieldReady={fieldReady}
   753→        loading={loading}
   754→        onTestFortyGuard={onTestFortyGuard}
   755→        onTestAI={onTestAI}
   756→      />
   757→
   758→      <div className="text-[10px] text-text-dimmed font-mono text-center">
   759→        FortyGuard Hackathon&apos;26
   760→      </div>
   761→    </div>
   762→  );
   763→}
