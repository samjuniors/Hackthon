    82→async function safeJsonFetch<T = Record<string, unknown>>(
    83→  url: string,
    84→  init?: RequestInit
    85→): Promise<{ ok: boolean; data: T | null }> {
    86→  try {
    87→    const res = await fetch(url, init);
    88→    const text = await res.text();
    89→    if (!text || !text.trim()) return { ok: res.ok, data: null };
    90→    try {
    91→      return { ok: res.ok, data: JSON.parse(text) as T };
    92→    } catch {
    93→      return { ok: false, data: null };
    94→    }
    95→  } catch {
    96→    return { ok: false, data: null };
    97→  }
    98→}

   477→
   478→  /**
   479→   * Location selection (Section 7 + 13):
   480→   *   - STATE/REGION result → geographic context ONLY: show boundary, dim the

   485→  const handleSelectLocation = useCallback((loc: NamedLocation) => {

   499→    const nextCenter = { latitude: loc.latitude, longitude: loc.longitude };

   610→  // Initial mount: trigger DEMO decision pipeline immediately + check health in parallel.

   634→      health: FortyGuardHealthResponse;
   635→      capability?: ProviderCapability | null;

   647→
   648→    safeJsonFetch('/api/health/ai', {

   651→      body: JSON.stringify({ preferredProvider: preferredProviderRef.current }),

   654→        const h = (data as { health: AIHealthResponse }).health;
   655→        setAiHealth(h);
   656→        setAiStatus(h.connected ? 'CONNECTED' : (h.configured ? 'ERROR' : 'UNKNOWN'));

   662→  }, []);

   664→  // React to mode changes (from Settings drawer or "Switch to LIVE" button).

   667→      didMountRef.current = true;
   668→      return; // initial mount handled above

   680→      selectedLocationRef.current = loc;
   681→      const nextCenter = { latitude: loc.latitude, longitude: loc.longitude };

   683→      aoiCenterRef.current = nextCenter;

   691→    temporalInputRef.current = newTemporal;

   696→      runDecisionPipeline(loc, aoi, newTemporal, FIXTURE_TIMEZONE, 'FIXTURE');
   697→    }
   698→    // LIVE: require explicit Generate — never auto-spend provider credits.
   699→     
   700→  }, [mode]);
   701→
   702→  // React to preferred-AI-provider changes (from Settings drawer).
   703→  useEffect(() => {
   704→    if (!didMountRef.current) return;
   705→    checkAIHealth();
   706→    if (jointDecisionRef.current) {
   707→      const activeScen = scenarioAnalysisRef.current?.scenarios?.find(
   708→        (s) => s.scenarioId === selectedScenarioIdRef.current
   709→      ) ?? scenarioAnalysisRef.current?.scenarios?.[0];
   710→      fetchExplanation(jointDecisionRef.current, activeScen);
   711→    }
   712→     
   713→  }, [prefs.preferredAIProvider]);
   714→
   715→  // React to AOI shape / size / resolution changes:
   716→  //   ALWAYS clear stale thermal data (Section 20).
   717→  //   FIXTURE auto-reruns (free). LIVE requires explicit Generate (credit safety).
   718→  useEffect(() => {
   719→    if (!didMountRef.current) return;
   720→    clearResults();
   721→    candidateSites.validateAgainstAoi(
   722→      createAoiFromSpan(aoiCenterRef.current, prefs.analysisAoiSpanMetres, prefs.analysisAreaShape)
   723→    );
   724→    if (modeRef.current === 'FIXTURE') {
   725→      const aoi = createAoiFromSpan(aoiCenterRef.current, prefs.analysisAoiSpanMetres, prefs.analysisAreaShape);
   726→      runDecisionPipeline(
   727→        selectedLocationRef.current,
   728→        aoi,
   729→        temporalInputRef.current,
   730→        FIXTURE_TIMEZONE,
   731→        'FIXTURE'
   732→      );
   733→    }
   734→     
   735→  }, [prefs.analysisAreaShape, prefs.analysisAoiSpanMetres, prefs.analysisResolution]);

   738→  // Derived values
   739→  // ───────────────────────────────────────────────────────────────────────────
   740→
   741→  const isFixtureMismatch = mode === 'FIXTURE' && !isLocationCoveredByFixture(selectedLocation);
   742→  const activeScenario = scenarioAnalysis?.scenarios?.find((s) => s.scenarioId === selectedScenarioId) ?? scenarioAnalysis?.scenarios?.[0];
   743→  const fieldReady = !!spatialField && !loading;
   744→  const altLocations = METROPOLITAN_LOCATIONS.filter((l) => !l.isDemoOnly).slice(0, 4);
   745→  const thermalCellCount = spatialField?.features?.length ?? 0;
   746→  const aiProvider = aiHealth?.provider;
   747→
   748→  // Candidates to display on the map + evaluate:
   749→  //   FIXTURE → the three DEMO CANDIDATES (application-defined points
   750→  //             evaluated against the captured field — not captured sites).
   751→  //   LIVE    → the user's real candidate sites only.

   753→    ? DEMO_CANDIDATE_SITES
   754→    : candidateSites.sites.map((s) => ({
   755→        locationId: s.locationId,
   756→        name: s.name,
   757→        location: s.location,

   917→              loading={loading}

   919→              analysisCenter={aoiCenter}
   920→              temporalInput={temporalInput}
   921→              timezone={displayTimezone}
   922→              rankedCandidates={spatialDecision?.rankedLocations.map((r) => ({
   923→                locationId: r.locationId,
   924→                name: r.name,
   925→                location: r.location,

   935→                regionDisplayName={regionName}

   971→                timezone={displayTimezone}

   996→
   997→            {/* Empty state before first run */}
   998→            {!jointDecision && !loading && !errorDetails && (
   999→              <div className="rounded-xl border border-border bg-surface-card px-5 py-12 text-center">
  1000→                <div className="text-2xl mb-2">🌡</div>

  1010→        </div>

  1015→    </main>
  1016→  );
