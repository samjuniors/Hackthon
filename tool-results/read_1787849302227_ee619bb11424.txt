   454→                    onSelectLocation={(loc) => {
   455→                      onAddSiteFromSearch(loc);
   456→                      setShowSiteSearch(false);
   457→                    }}
   458→                    onSwitchToLive={undefined}
   459→                    compact
   460→                    activeStateFilter={activeStateFilter}
   461→                  />
   462→                  <p className="text-[9px] text-text-dimmed mt-1">
   463→                    Only sites inside the analysis area can be added.
   464→                  </p>
   465→                </div>
   466→              )}
   467→
   468→              {/* Site list */}
   469→              {candidateSites.length === 0 ? (
   470→                <p className="text-[10px] text-text-muted leading-relaxed" data-testid="no-candidate-sites">
   471→                  No candidate sites yet. LIVE never fabricates sites — add one on the map or from search, then Generate.
   472→                </p>
   473→              ) : (
   474→                <ul className="space-y-1.5 max-h-40 overflow-y-auto" data-testid="candidate-sites-list">
   475→                  {candidateSites.map((site, idx) => (
   476→                    <li
   477→                      key={site.locationId}
   478→                      className={`rounded-md border px-2.5 py-1.5 ${
   479→                        site.outsideAoi
   480→                          ? 'border-red-400/60 bg-red-400/5'
   481→                          : 'border-border bg-surface-deep'
   482→                      }`}
   483→                    >
   484→                      <div className="flex items-center gap-2">
   485→                        <span
   486→                          className="w-4 h-4 rounded-full flex items-center justify-center text-[9px] font-mono font-bold text-white flex-shrink-0 shadow-sm"
   487→                          style={{ backgroundColor: getCandidateColor(idx) }}
   488→                          title={`Site ${idx + 1}`}
   489→                        >
   490→                          {idx + 1}
   491→                        </span>
   492→                        {renamingId === site.locationId ? (
   493→                          <input
   494→                            autoFocus
   495→                            value={renameValue}
   496→                            onChange={(e) => setRenameValue(e.target.value)}
   497→                            onBlur={() => {
   498→                              if (renameValue.trim()) onRenameSite(site.locationId, renameValue.trim());
   499→                              setRenamingId(null);
   500→                            }}
   501→                            onKeyDown={(e) => {
   502→                              if (e.key === 'Enter') {
   503→                                if (renameValue.trim()) onRenameSite(site.locationId, renameValue.trim());
   504→                                setRenamingId(null);
   505→                              }
   506→                            }}
   507→                            className="flex-1 min-w-0 bg-surface-elevated border border-accent-cyan/50 rounded px-1.5 py-0.5 text-xs text-text-primary focus:outline-none"
   508→                          />
   509→                        ) : (
   510→                          <button
   511→                            type="button"
   512→                            onClick={() => {
   513→                              setRenamingId(site.locationId);
   514→                              setRenameValue(site.name);
   515→                            }}
   516→                            className="flex-1 min-w-0 text-left text-xs text-text-secondary hover:text-text-primary transition-colors truncate"
   517→                            title="Click to rename"
   518→                          >
   519→                            {site.name}
   520→                          </button>
   521→                        )}
   522→                        <span
   523→                          className="text-[8px] font-mono uppercase text-text-dimmed flex-shrink-0"
   524→                          title={`Added via ${site.origin}`}
   525→                        >
   526→                          {site.origin === 'map-click' ? 'map' : site.origin}
   527→                        </span>
   528→                        <button
   529→                          type="button"
   530→                          onClick={() => onRemoveSite(site.locationId)}
   531→                          className="text-text-dimmed hover:text-red-400 transition-colors text-xs flex-shrink-0"
   532→                          title="Remove site"
   533→                        >
   534→                          ✕
   535→                        </button>
   536→                      </div>
   537→                      {site.outsideAoi && (
   538→                        <div className="text-[9px] text-red-400 mt-0.5">
   539→                          Outside the analysis area — move it inside or drag the AOI to cover it.
   540→                        </div>
   541→                      )}
   542→                    </li>
   543→                  ))}
   544→                </ul>
   545→              )}
   546→              {outsideSiteCount > 0 && (
   547→                <p className="text-[9px] text-red-400" data-testid="outside-sites-warning">
   548→                  {outsideSiteCount} site{outsideSiteCount > 1 ? 's' : ''} outside the analysis area — fix before Generate.
   549→                </p>
   550→              )}
   551→            </div>
   552→          )}
   553→        </div>
   554→
   555→        <div className="border-t border-border" />
   556→
   557→        {/* ── WHEN — explicit date + time window (Section 14) ── */}
   558→        <div data-testid="when-section">
   559→          <div className="flex items-center justify-between mb-2">
   560→            <span className="text-sm font-medium text-text-secondary">WHEN</span>
   561→            {isFixtureAnchored ? (
   562→              <span
   563→                className="px-2 py-0.5 rounded text-[9px] font-mono font-bold border"
   564→                style={{
   565→                  background: 'var(--accent-amber-bg)',
   566→                  color: 'var(--accent-amber)',
   567→                  borderColor: 'var(--accent-amber)',
   568→                }}
   569→                title={FIXTURE_TEMPORAL_METADATA.captureLabel}
   570→              >
   571→                Fixture capture
   572→              </span>
   573→            ) : (
   574→              <span className="text-[10px] font-mono text-text-dimmed">{tz || 'UTC'}</span>
   575→            )}
   576→          </div>
   577→
   578→          {/* EVALUATION WINDOW selector — a UI concept, NOT a provider
   579→              filter_type. The verified wire contract: every evaluated hour is
   580→              its own single-hour FortyGuard request (filter_type: 1). */}
   581→          <div className="mb-2.5">
   582→            <div className="flex items-center justify-between mb-1.5">
   583→              <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">
   584→                Evaluation Window
   585→              </span>
   586→            </div>
   587→            <div className="grid grid-cols-2 gap-1.5">
   588→              {TIME_MODE_OPTIONS.map((opt) => {
   589→                const active = temporalInput.timeMode === opt.value;
   590→                return (
   591→                  <button
   592→                    key={opt.value}
   593→                    data-testid={`evaluation-window-${opt.value}`}
   594→                    onClick={() => handleTimeModeChange(opt.value)}
   595→                    className={`min-h-[38px] rounded-lg text-[10px] transition-all border leading-tight px-1 ${
   596→                      active
   597→                        ? 'border-slate-900 bg-slate-900 text-white dark:border-cyan-400 dark:bg-cyan-500/20 dark:text-cyan-300 font-bold shadow-xs'
   598→                        : 'border-slate-300 dark:border-slate-800 bg-white dark:bg-slate-900/60 text-slate-700 dark:text-slate-300 hover:bg-slate-100 dark:hover:bg-slate-800 hover:text-slate-900 dark:hover:text-slate-100 font-medium'
   599→                    }`}
   600→                    aria-pressed={active}
   601→                    title={opt.description}
   602→                  >
   603→                    {opt.label}
   604→                  </button>
   605→                );
   606→              })}
   607→            </div>
   608→            <p className="text-[9px] text-text-dimmed mt-1 leading-relaxed">
   609→              A time range is evaluated as a sequence of hourly FortyGuard requests — one request per hour.
   610→            </p>
   611→          </div>
   612→
   613→          {/* Date input (always visible) */}
   614→          <label className="block mb-2">
   615→            <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">Date</span>
   616→            <input
   617→              type="date"
   618→              value={temporalInput.date}
   619→              onChange={(e) => update({ date: e.target.value })}
   620→              disabled={isFixtureAnchored}
   621→              className={`mt-1 w-full min-h-[42px] rounded-lg border bg-white dark:bg-slate-900 px-3 text-sm font-mono text-slate-900 dark:text-slate-100 font-medium focus:outline-none focus:border-slate-900 dark:focus:border-cyan-400 focus:ring-1 focus:ring-slate-900/20 transition-colors ${
   622→                dateValid ? 'border-slate-300 dark:border-slate-700' : 'border-rose-500'
   623→              } ${isFixtureAnchored ? 'opacity-70 cursor-not-allowed' : ''}`}
   624→              aria-label="Analysis date"
   625→            />
   626→            {/* LIVE date hint — honest about model completion, no invented
   627→                "guaranteed valid" date, no silent retry / DEMO fallback. */}
   628→            {mode === 'LIVE' && (
   629→              <span className="block text-[9px] text-text-dimmed mt-1 leading-relaxed" data-testid="live-date-hint">
   630→                Pick the date explicitly. Recently requested periods may have no completed FortyGuard model yet (the
   631→                provider then returns an empty thermal field — reported verbatim, never retried with another date).
   632→                Documented forecast support: up to +12h ahead.
   633→              </span>
   634→            )}
   635→          </label>
   636→
   637→          {/* Start / End time inputs */}
   638→          <div className={`grid ${isSingleHour ? 'grid-cols-1' : 'grid-cols-2'} gap-2 mb-2`}>
   639→            <label className="block">
   640→              <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">
   641→                {isSingleHour ? 'Hour' : 'Start'}
   642→              </span>
   643→              <input
   644→                type="time"
   645→                value={temporalInput.startTime}
   646→                onChange={(e) => update({ startTime: e.target.value })}
   647→                disabled={isFixtureAnchored}
   648→                className={`mt-1 w-full min-h-[42px] rounded-lg border bg-white dark:bg-slate-900 px-3 text-sm font-mono text-slate-900 dark:text-slate-100 font-medium focus:outline-none focus:border-slate-900 dark:focus:border-cyan-400 focus:ring-1 focus:ring-slate-900/20 transition-colors ${
   649→                  startValid ? 'border-slate-300 dark:border-slate-700' : 'border-rose-500'
   650→                } ${isFixtureAnchored ? 'opacity-70 cursor-not-allowed' : ''}`}
   651→                aria-label="Analysis start time"
   652→              />
   653→            </label>
   654→            {!isSingleHour && (
   655→              <label className="block">
   656→                <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">End</span>
   657→                <input
   658→                  type="time"
   659→                  value={temporalInput.endTime}
   660→                  onChange={(e) => update({ endTime: e.target.value })}
   661→                  disabled={isFixtureAnchored}
   662→                  className={`mt-1 w-full min-h-[42px] rounded-lg border bg-white dark:bg-slate-900 px-3 text-sm font-mono text-slate-900 dark:text-slate-100 font-medium focus:outline-none focus:border-slate-900 dark:focus:border-cyan-400 focus:ring-1 focus:ring-slate-900/20 transition-colors ${
   663→                    endValid ? 'border-slate-300 dark:border-slate-700' : 'border-rose-500'
   664→                  } ${isFixtureAnchored ? 'opacity-70 cursor-not-allowed' : ''}`}
   665→                  aria-label="Analysis end time"
   666→                />
   667→              </label>
   668→            )}
   669→          </div>
   670→
   671→          {/* Derived duration (read-only) */}
   672→          <div className="flex items-center justify-between rounded-lg bg-slate-100 dark:bg-slate-900/80 px-3 py-2 border border-slate-200 dark:border-slate-800">
   673→            <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">
   674→              Duration
   675→            </span>
   676→            <span
   677→              className="text-base font-black font-mono text-slate-950 dark:text-cyan-300"
   678→              data-testid="duration-display"
   679→            >
   680→              {derivedDuration}h
   681→            </span>
   682→          </div>
   683→
   684→          {/* Inline validation hint */}
   685→          {!allValid && (
   686→            <p className="text-[10px] text-rose-600 dark:text-rose-400 mt-1.5 font-medium">
   687→              {!dateValid && 'Enter a valid date (YYYY-MM-DD). '}
   688→              {!rangeValid && 'End time must be after start time.'}
   689→            </p>
   690→          )}
   691→
   692→          {/* Honest future-date note */}
   693→          {mode === 'LIVE' && isFutureDate && (
   694→            <p className="text-[10px] mt-1.5 font-medium text-amber-700 dark:text-amber-300">
   695→              Future date selected — subject to FortyGuard forecast availability. The provider reports any unsupported window verbatim.
   696→            </p>
   697→          )}
   698→
   699→          {/* LIVE billing disclosure — the engine sends ONE hourly provider
   700→              request per evaluated hour; make that explicit BEFORE Generate.
   701→              Conservative wording: no exact credit cost is claimed. */}
   702→          {mode === 'LIVE' && allValid && (
   703→            <div
   704→              className="rounded-lg px-3 py-2 mt-2 border border-emerald-300 dark:border-emerald-800/80 bg-emerald-50/80 dark:bg-emerald-950/20"
   705→              data-testid="live-request-disclosure"
   706→            >
   707→              <div className="flex items-center justify-between">
   708→                <span className="text-[9px] font-bold uppercase tracking-widest text-emerald-800 dark:text-emerald-400">
   709→                  LIVE · {derivedDuration}-hour evaluation
   710→                </span>
   711→                <span className="text-[10px] font-mono font-bold text-emerald-800 dark:text-emerald-300">
   712→                  {liveHourlyRequestCount} FortyGuard hourly request{liveHourlyRequestCount > 1 ? 's' : ''}
   713→                </span>
   714→              </div>
   715→              <p className="text-[9px] text-slate-600 dark:text-slate-400 mt-0.5 leading-relaxed">
   716→                Each evaluated hour is submitted as its own single-hour FortyGuard /v1/heatmap request (cached results are
   717→                reused, not re-billed). Repeat requests may consume provider credits.
   718→              </p>
   719→            </div>
   720→          )}
   721→
   722→          {/* Human-readable window preview */}
   723→          {allValid && (
   724→            <p className="text-[10px] font-mono text-slate-600 dark:text-slate-400 mt-2 leading-relaxed">
   725→              {formatTemporalForHeader(temporalInput, tz)}
   726→            </p>
   727→          )}
   728→        </div>
   729→
   730→        {/* Selected Analysis Area indicator — coordinates track AOI drag */}
   731→        <div
   732→          className="rounded-lg p-3 bg-slate-50 dark:bg-slate-900/60 border border-slate-200 dark:border-slate-800 space-y-1"
   733→          data-testid="active-analysis-location-indicator"
   734→        >
   735→          <div className="flex items-center justify-between">
   736→            <span className="text-[9px] font-bold uppercase tracking-widest text-text-dimmed">Selected Analysis Area</span>
   737→            <span
   738→              className="px-2 py-0.5 rounded text-[10px] font-bold border"
   739→              style={mode === 'LIVE'
   740→                ? { background: 'var(--accent-emerald-bg)', color: 'var(--accent-emerald-text)', borderColor: 'var(--accent-emerald)' }
   741→                : { background: 'var(--accent-amber-bg)', color: 'var(--accent-amber-text)', borderColor: 'var(--accent-amber)' }
   742→              }
   743→              data-testid="analysis-mode-badge"
   744→            >
   745→              {mode === 'LIVE' ? 'LIVE · FortyGuard' : 'DEMO · Captured FortyGuard'}
   746→            </span>
   747→          </div>
   748→          <div className="text-sm font-bold text-slate-900 dark:text-slate-100 leading-tight" data-testid="active-analysis-location-name">
   749→            {selectedLocation ? selectedLocation.name : 'No location selected'}
   750→          </div>
   751→          {selectedLocation && centerCoords && (
   752→            <div className="text-[11px] font-mono flex items-center justify-between text-slate-700 dark:text-cyan-300" data-testid="active-analysis-location-coords">
   753→              <span className="font-semibold">{centerCoords.latitude.toFixed(4)}°, {centerCoords.longitude.toFixed(4)}°</span>
   754→              <span className="text-slate-500 font-sans text-[10px]">
   755→                {selectedLocation.city ? `${selectedLocation.city}, ${selectedLocation.state || selectedLocation.country}` : selectedLocation.state || ''}
   756→              </span>
   757→            </div>
   758→          )}
   759→        </div>
   760→
   761→        {/* Generate button — solid Apple / Linear / Stripe high contrast styling */}
   762→        <motion.button
   763→          whileTap={{ scale: 0.99 }}
   764→          disabled={loading || generateDisabled}
   765→          onClick={onGenerate}
   766→          data-testid="recalculate-decision-btn"
   767→          title={generateDisabled && !loading ? generateDisabledReason : undefined}
   768→          className={`w-full min-h-[48px] rounded-xl text-sm font-bold transition-all flex items-center justify-center gap-2 border shadow-xs ${
   769→            loading || generateDisabled
   770→              ? 'bg-slate-100 dark:bg-slate-800/60 border-slate-200 dark:border-slate-800 text-slate-400 dark:text-slate-500 cursor-not-allowed'
   771→              : mode === 'LIVE'
   772→                ? 'bg-slate-900 hover:bg-slate-800 dark:bg-emerald-600 dark:hover:bg-emerald-500 text-white border-slate-900 dark:border-emerald-500 cursor-pointer active:bg-black'
   773→                : 'bg-slate-900 hover:bg-slate-800 dark:bg-sky-600 dark:hover:bg-sky-500 text-white border-slate-900 dark:border-sky-500 cursor-pointer active:bg-black'
   774→          }`}
   775→        >
   776→          {loading ? (
   777→            <span className="flex items-center gap-2">
   778→              <span className="w-4 h-4 border-2 border-white/40 border-t-white rounded-full animate-spin" />
   779→              Generating thermal field…
   780→            </span>
   781→          ) : (
   782→            `⚡ Generate Thermal Field (${mode === 'LIVE' ? 'LIVE' : 'DEMO'})`
   783→          )}
   784→        </motion.button>
   785→
   786→        {/* Inline reason when Generate is blocked (validation feedback) */}
   787→        {generateDisabled && !loading && generateDisabledReason && (
   788→          <p
   789→            className="text-[10px] leading-relaxed text-center -mt-1 font-medium text-amber-700 dark:text-amber-400"
   790→            data-testid="generate-blocked-reason"
   791→          >
   792→            ⚠ {generateDisabledReason}
   793→          </p>
   794→        )}
   795→      </div>
   796→
   797→      {/* ── SYSTEM STATUS ── */}
   798→      <SystemStatus
   799→        mode={mode}
   800→        fortyGuardStatus={fortyGuardStatus}
   801→        aiStatus={aiStatus}
   802→        aiProvider={aiProvider}
   803→        fieldReady={fieldReady}
   804→        loading={loading}
   805→        onTestFortyGuard={onTestFortyGuard}
   806→        onTestAI={onTestAI}
   807→      />
   808→
   809→      <div className="text-[10px] text-text-dimmed font-mono text-center">
   810→        FortyGuard Hackathon&apos;26
   811→      </div>
   812→    </div>
   813→  );
   814→}
