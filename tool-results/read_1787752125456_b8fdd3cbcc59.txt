   493→                dateValid ? 'border-border' : 'border-red-400'
   494→              } ${isFixtureAnchored ? 'opacity-70 cursor-not-allowed' : ''}`}
   495→              aria-label="Analysis date"
   496→            />
   497→          </label>
   498→
   499→          {/* Start / End time inputs */}
   500→          <div className={`grid ${isSingleHour ? 'grid-cols-1' : 'grid-cols-2'} gap-2 mb-2`}>
   501→            <label className="block">
   502→              <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">
   503→                {isSingleHour ? 'Hour' : 'Start'}
   504→              </span>
   505→              <input
   506→                type="time"
   507→                value={temporalInput.startTime}
   508→                onChange={(e) => update({ startTime: e.target.value })}
   509→                disabled={isFixtureAnchored || isSingleDay}
   510→                className={`mt-1 w-full h-10 rounded-lg border bg-surface-elevated px-3 text-sm font-mono text-text-primary focus:outline-none focus:border-accent-cyan transition-colors ${
   511→                  startValid ? 'border-border' : 'border-red-400'
   512→                } ${isFixtureAnchored || isSingleDay ? 'opacity-70 cursor-not-allowed' : ''}`}
   513→                aria-label="Analysis start time"
   514→              />
   515→            </label>
   516→            {!isSingleHour && (
   517→              <label className="block">
   518→                <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">End</span>
   519→                <input
   520→                  type="time"
   521→                  value={temporalInput.endTime}
   522→                  onChange={(e) => update({ endTime: e.target.value })}
   523→                  disabled={isFixtureAnchored || isSingleDay}
   524→                  className={`mt-1 w-full h-10 rounded-lg border bg-surface-elevated px-3 text-sm font-mono text-text-primary focus:outline-none focus:border-accent-cyan transition-colors ${
   525→                    endValid ? 'border-border' : 'border-red-400'
   526→                  } ${isFixtureAnchored || isSingleDay ? 'opacity-70 cursor-not-allowed' : ''}`}
   527→                  aria-label="Analysis end time"
   528→                />
   529→              </label>
   530→            )}
   531→          </div>
   532→
   533→          {/* Single Day: window-length selector */}
   534→          {isSingleDay && (
   535→            <div className="mb-2">
   536→              <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">
   537→                Window length to find
   538→              </span>
   539→              <div className="grid grid-cols-3 gap-1.5 mt-1">
   540→                {([2, 3, 4] as const).map((h) => {
   541→                  const active = (temporalInput.dayWindowHours ?? 3) === h;
   542→                  return (
   543→                    <button
   544→                      key={h}
   545→                      onClick={() => handleDayWindowChange(h)}
   546→                      className={`min-h-[34px] rounded-md text-[11px] font-bold font-mono transition-all border ${
   547→                        active
   548→                          ? 'border-accent-cyan bg-accent-cyan-bg text-accent-cyan'
   549→                          : 'border-border bg-surface-elevated text-text-muted hover:text-text-primary hover:bg-surface-deep'
   550→                      }`}
   551→                      aria-pressed={active}
   552→                    >
   553→                      {h}h
   554→                    </button>
   555→                  );
   556→                })}
   557→              </div>
   558→            </div>
   559→          )}
   560→
   561→          {/* Derived duration (read-only) */}
   562→          <div className="flex items-center justify-between rounded-lg bg-surface-deep px-3 py-2 border border-border">
   563→            <span className="text-[10px] font-bold uppercase tracking-widest text-text-dimmed">
   564→              Duration
   565→            </span>
   566→            <span
   567→              className="text-base font-black font-mono"
   568→              style={{ color: 'var(--accent-cyan)' }}
   569→              data-testid="duration-display"
   570→            >
   571→              {derivedDuration}h
   572→            </span>
   573→          </div>
   574→
   575→          {/* Inline validation hint */}
   576→          {!allValid && (
   577→            <p className="text-[10px] text-red-400 mt-1.5">
   578→              {!dateValid && 'Enter a valid date (YYYY-MM-DD). '}
   579→              {!rangeValid && 'End time must be after start time.'}
   580→            </p>
   581→          )}
   582→
   583→          {/* Honest future-date note */}
   584→          {mode === 'LIVE' && isFutureDate && (
   585→            <p className="text-[10px] mt-1.5" style={{ color: 'var(--accent-amber)' }}>
   586→              Future date selected — subject to FortyGuard forecast availability. The provider reports any unsupported window verbatim.
   587→            </p>
   588→          )}
   589→
   590→          {/* Human-readable window preview */}
   591→          {allValid && (
   592→            <p className="text-[10px] font-mono text-text-muted mt-2 leading-relaxed">
   593→              {formatTemporalForHeader(temporalInput, tz)}
   594→            </p>
   595→          )}
   596→        </div>
   597→
   598→        {/* Selected Analysis Area indicator — coordinates track AOI drag */}
   599→        <div
   600→          className="rounded-lg p-3 bg-surface-deep border border-border space-y-1"
   601→          data-testid="active-analysis-location-indicator"
   602→        >
   603→          <div className="flex items-center justify-between">
   604→            <span className="text-[9px] font-bold uppercase tracking-widest text-text-dimmed">Selected Analysis Area</span>
   605→            <span
   606→              className="px-2 py-0.5 rounded text-[10px] font-bold border"
   607→              style={mode === 'LIVE'
   608→                ? { background: 'var(--accent-cyan-bg)', color: 'var(--accent-cyan)', borderColor: 'var(--accent-cyan)' }
   609→                : { background: 'var(--accent-amber-bg)', color: 'var(--accent-amber)', borderColor: 'var(--accent-amber)' }
   610→              }
   611→              data-testid="analysis-mode-badge"
   612→            >
   613→              {mode === 'LIVE' ? 'LIVE · FortyGuard' : 'DEMO · Captured FortyGuard'}
   614→            </span>
   615→          </div>
   616→          <div className="text-sm font-bold text-text-primary leading-tight" data-testid="active-analysis-location-name">
   617→            {selectedLocation.name}
   618→          </div>
   619→          <div className="text-[11px] font-mono flex items-center justify-between" style={{ color: 'var(--accent-cyan)' }} data-testid="active-analysis-location-coords">
   620→            <span>{centerCoords.latitude.toFixed(4)}°, {centerCoords.longitude.toFixed(4)}°</span>
   621→            <span className="text-text-muted font-sans text-[10px]">
   622→              {selectedLocation.city ? `${selectedLocation.city}, ${selectedLocation.state || selectedLocation.country}` : selectedLocation.state || ''}
   623→            </span>
   624→          </div>
   625→        </div>
   626→
   627→        {/* Generate button */}
   628→        <motion.button
   629→          whileTap={{ scale: 0.99 }}
   630→          disabled={loading}
   631→          onClick={onGenerate}
   632→          data-testid="recalculate-decision-btn"
   633→          className={`w-full h-12 rounded-xl text-sm font-bold transition-all flex items-center justify-center gap-2 ${
   634→            loading ? 'bg-surface-elevated text-text-dimmed cursor-not-allowed' : 'text-white hover:scale-[1.01] active:scale-[0.99]'
   635→          }`}
   636→          style={loading ? {} : {
   637→            background: mode === 'LIVE'
   638→              ? 'linear-gradient(135deg, var(--accent-emerald), #0d9488)'
   639→              : 'linear-gradient(135deg, var(--accent-cyan), #0284c7)',
   640→            boxShadow: mode === 'LIVE'
   641→              ? '0 4px 16px rgba(5,150,105,0.3)'
   642→              : '0 4px 16px rgba(14,165,233,0.3)',
   643→          }}
   644→        >
   645→          {loading ? (
   646→            <span className="flex items-center gap-2">
   647→              <span className="w-4 h-4 border-2 border-text-dimmed rounded-full animate-spin" style={{ borderTopColor: 'var(--accent-cyan)' }} />
   648→              Generating thermal field…
   649→            </span>
   650→          ) : (
   651→            `⚡ Generate Thermal Field (${mode === 'LIVE' ? 'LIVE' : 'DEMO'})`
   652→          )}
   653→        </motion.button>
   654→      </div>
   655→
   656→      {/* ── SYSTEM STATUS ── */}
   657→      <SystemStatus
   658→        mode={mode}
   659→        fortyGuardStatus={fortyGuardStatus}
   660→        aiStatus={aiStatus}
   661→        aiProvider={aiProvider}
   662→        fieldReady={fieldReady}
   663→        loading={loading}
   664→        onTestFortyGuard={onTestFortyGuard}
   665→        onTestAI={onTestAI}
   666→      />
   667→
   668→      <div className="text-[10px] text-text-dimmed font-mono text-center">
   669→        FortyGuard Hackathon&apos;26
   670→      </div>
   671→    </div>
   672→  );
   673→}

   675→export default ControlRail;
