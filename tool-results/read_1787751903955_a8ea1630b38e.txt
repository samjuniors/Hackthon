    76→async function safeJsonFetch<T = Record<string, unknown>>(
    77→  url: string,
    78→  init?: RequestInit
    79→): Promise<{ ok: boolean; data: T | null }> {
    80→  try {
    81→    const res = await fetch(url, init);
    82→    const text = await res.text();
    83→    if (!text || !text.trim()) return { ok: res.ok, data: null };
    84→    try {
    85→      return { ok: res.ok, data: JSON.parse(text) as T };
    86→    } catch {
    87→      return { ok: false, data: null };
    88→    }
    89→  } catch {
    90→    return { ok: false, data: null };
    91→  }
    92→}

   611→      success: boolean;
   612→      health: FortyGuardHealthResponse;
   613→      capability?: ProviderCapability | null;

   626→    safeJsonFetch('/api/health/ai', {

   634→        setAiStatus(h.connected ? 'CONNECTED' : (h.configured ? 'ERROR' : 'UNKNOWN'));

   654→    let loc = selectedLocationRef.current;

   659→      const nextCenter = { latitude: loc.latitude, longitude: loc.longitude };
   660→      setAoiCenter(nextCenter);
   661→      aoiCenterRef.current = nextCenter;

   669→    temporalInputRef.current = newTemporal;
   670→
   671→    if (newMode === 'FIXTURE') {
   672→      // DEMO is free (captured fixture) — auto-run for instant feedback.
   673→      const aoi = createAoiFromSpan(aoiCenterRef.current, prefs.analysisAoiSpanMetres, prefs.analysisAreaShape);
   674→      runDecisionPipeline(loc, aoi, newTemporal, loc.timezone || 'UTC', 'FIXTURE');
   675→    }
   676→    // LIVE: require explicit Generate — never auto-spend provider credits.
   677→     
   678→  }, [mode]);
   679→
   680→  // React to preferred-AI-provider changes (from Settings drawer).
   681→  useEffect(() => {
   682→    if (!didMountRef.current) return;
   683→    checkAIHealth();
   684→    if (jointDecisionRef.current) {
   685→      const activeScen = scenarioAnalysisRef.current?.scenarios?.find(
   686→        (s) => s.scenarioId === selectedScenarioIdRef.current
   687→      ) ?? scenarioAnalysisRef.current?.scenarios?.[0];
   688→      fetchExplanation(jointDecisionRef.current, activeScen);
   689→    }
   690→     
   691→  }, [prefs.preferredAIProvider]);
   692→
   693→  // React to AOI shape / size / resolution changes:
   694→  //   ALWAYS clear stale thermal data (Section 20).
   695→  //   FIXTURE auto-reruns (free). LIVE requires explicit Generate (credit safety).
   696→  useEffect(() => {
   697→    if (!didMountRef.current) return;
   698→    clearResults();
   699→    candidateSites.validateAgainstAoi(
   700→      createAoiFromSpan(aoiCenterRef.current, prefs.analysisAoiSpanMetres, prefs.analysisAreaShape)
   701→    );
   702→    if (modeRef.current === 'FIXTURE') {
   703→      const aoi = createAoiFromSpan(aoiCenterRef.current, prefs.analysisAoiSpanMetres, prefs.analysisAreaShape);
   704→      runDecisionPipeline(
   705→        selectedLocationRef.current,
   706→        aoi,
   707→        temporalInputRef.current,
   708→        selectedLocationRef.current.timezone || 'UTC',
   709→        'FIXTURE'
   710→      );
   711→    }
   712→     
   713→  }, [prefs.analysisAreaShape, prefs.analysisAoiSpanMetres, prefs.analysisResolution]);

   718→
   719→  const isFixtureMismatch = mode === 'FIXTURE' && !isLocationCoveredByFixture(selectedLocation);
   720→  const activeScenario = scenarioAnalysis?.scenarios?.find((s) => s.scenarioId === selectedScenarioId) ?? scenarioAnalysis?.scenarios?.[0];
   721→  const fieldReady = !!spatialField && !loading;
   722→  const altLocations = METROPOLITAN_LOCATIONS.filter((l) => !l.isDemoOnly).slice(0, 4);
   723→  const thermalCellCount = spatialField?.features?.length ?? 0;
   724→  const aiProvider = aiHealth?.provider;
   725→

   727→  //   FIXTURE → the three ACTUAL captured Manhattan demo sites.

   730→    ? CAPTURED_DEMO_SITES

   732→        locationId: s.locationId,
   733→        name: s.name,
   734→        location: s.location,

   765→      selectedLocationRef.current.timezone || 'UTC',

   769→            locationId: s.locationId,

   781→      temporalInputRef.current,

   892→              temporalInput={temporalInput}

   894→              rankedCandidates={spatialDecision?.rankedLocations.map((r) => ({
   895→                locationId: r.locationId,
   896→                name: r.name,
   897→                location: r.location,
   898→              }))}
   899→              recommendedLocationId={spatialDecision?.recommendedLocation.locationId}
   900→            >
   901→              <ThermalMap
   902→                location={aoiCenter}
   903→                locationState={selectedLocation.state}
   904→                locationName={selectedLocation.name}
   905→                regionBoundary={regionBoundary}
   906→                regionMask={regionMask}

   940→                jointDecision={jointDecision}

   945→

   950→                selectedScenarioId={selectedScenarioId}

   953→                timezone={selectedLocation.timezone}

   956→
   957→            {/* 5. GROUNDED AI EXPLANATION (subdued — not the hero) */}

   967→
   968→            {/* Empty state before first run */}

   971→                <div className="text-2xl mb-2">🌡</div>

   977→                </p>
   978→              </div>
   979→            )}
   980→          </div>
   981→        </div>
   982→      </div>
   983→
   984→      {/* SETTINGS DRAWER — provider capability + diagnostics */}
   985→      <SettingsDrawer open={settingsOpen} onOpenChange={setSettingsOpen} capability={capability} />
   986→    </main>
   987→  );
   988→}
